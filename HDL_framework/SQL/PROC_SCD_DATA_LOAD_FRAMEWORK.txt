						||VAR_TARGET_DATASET
						||'.'
						||VAR_TARGET_TABLE

						||VAR_TARGET_COLUMN

						||VAR_SOURCE_PROJECT_ID
						||'.'
						||VAR_SOURCE_DATASET

					    ||'\''
 					    ||' AND STATUS = '
						||'\''
						||'SUCCESS'
						||'\''
					    || ''')''';

	  EXECUTE IMMEDIATE VAR_SQL_STR;

						||VAR_SOURCE_DATASET

    					||VAR_SOURCE_TABLE;*/
						
      SET VAR_SQL_STR = VAR_SQL_STR
	                    ||' AS SELECT '

                        ||VAR_TARGET_PROJECT_ID

						||VAR_TARGET_DATASET
    					||'.'
    					||VAR_TARGET_TABLE
    					||' RENAME  TO '
    					||VAR_TARGET_TABLE
    					||'_BACKUP_'
    					||FORMAT_DATE('%Y%m%d', RUN_DATE);

                        ||VAR_TARGET_PROJECT_ID
						||'.'
						||VAR_TARGET_DATASET
    					||'.'
    					||VAR_TARGET_TABLE
    					||'_TEMP'

    					||VAR_TARGET_TABLE;

      THEN

                          ||VAR_TARGET_PROJECT_ID
						  ||'.'
						  ||VAR_TARGET_DATASET
    	      			  ||'.'
       					  ||VAR_TARGET_TABLE
    					  ||'_BACKUP_'
    					  ||FORMAT_DATE('%Y%m%d', RUN_DATE);

        EXECUTE IMMEDIATE VAR_SQL_STR;
        SET RECORD_COUNT = IFNULL(@@ROW_COUNT,0);

      ELSE

                          ||VAR_TARGET_PROJECT_ID
						  ||'.'
						  ||VAR_TARGET_DATASET
    	      			  ||'.'
    					  ||VAR_TARGET_TABLE;

                          ||VAR_TARGET_PROJECT_ID
						  ||'.'
						  ||VAR_TARGET_DATASET
    		              ||'.'
    				      ||VAR_TARGET_TABLE
    					  ||'_BACKUP_'
    					  ||FORMAT_DATE('%Y%m%d', RUN_DATE)
                          ||' RENAME TO '

    	select "LINE 279";

	  EXECUTE IMMEDIATE 'DROP TABLE IF EXISTS '||VAR_TARGET_PROJECT_ID||'.'||VAR_TARGET_DATASET||'.'||VAR_TARGET_TABLE||'_BACKUP_'||FORMAT_DATE('%Y%m%d', RUN_DATE);

				   INFO AS 
				          (SELECT COLUMN_NAME,
					              CASE
					                WHEN DATA_TYPE IN ('INT64','INTEGER', 'NUMERIC', 'BIGNUMERIC','BIGNUMERIC(38, 19)','FLOAT64') 
									THEN 'IFNULL(CAST('||COLUMN_NAME||' AS STRING),\'-1\')'
					                WHEN DATA_TYPE LIKE 'STRING%' THEN 'IFNULL('||COLUMN_NAME||',\'-1\')'
					                WHEN DATA_TYPE='DATE' THEN 'IFNULL('||COLUMN_NAME||',\'1900-01-01\')'
					                WHEN DATA_TYPE='TIME' THEN 'IFNULL('||COLUMN_NAME||',\'00:00:00\')'
					                WHEN DATA_TYPE='DATETIME' THEN 'IFNULL('||COLUMN_NAME||',\'1900-01-01T00:00:00\')'
					                WHEN DATA_TYPE='TIMESTAMP' THEN 'IFNULL('||COLUMN_NAME||',\'1900-01-01 00:00:00.000000 UTC\')'
					                ELSE 'IFNULL('||COLUMN_NAME||',-1)'
				                   END
					              AS ABI_COLS
				             FROM `region-us`.INFORMATION_SCHEMA.COLUMNS
				            WHERE  '`'||table_catalog||'`'|| '.'|| TABLE_SCHEMA||'.'||TABLE_NAME = TARGET_TABLE_PARAM

					      ),
				   TAB1 AS 
				         (SELECT TRIM(COLS) AS COLUMN_NAME
				            FROM TAB
				           CROSS JOIN UNNEST(TAB.TARGET_COLS) AS COLS
						 )
				   SELECT STRING_AGG(INFO.ABI_COLS,'||\'~\'||')
				     FROM INFO
				    INNER JOIN TAB1
				       ON INFO.COLUMN_NAME = TAB1.COLUMN_NAME
				  );

		SET SP_STATUS =1 ;
		SELECT RECORD_COUNT, SP_STATUS;
		RETURN;	
	  END IF;

	  SET VAR_BUS_KEY_COL =(WITH 
	                        ARR AS 
							      (SELECT SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
								     FROM `PROJECT_ID_31`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
									WHERE '`'||TARGET_PROJECT_ID||'`.'||TARGET_DATASET||'.'||TARGET_TABLE = TARGET_TABLE_PARAM
								  ) 
							SELECT STRING_AGG('MAIN.'||KEYCOLUMNS||' = '||'WRK.'||KEYCOLUMNS,' AND ') CODN
							  FROM ARR
							 CROSS JOIN UNNEST(KEYCOLUMNS) AS KEYCOLUMNS
						   );
				
	  SELECT VAR_BUS_KEY_COL; 

  	  /*SETTING UPDATE_COLS RUNTIME */
	  SET UPDATE_COLS =(WITH
						 ARR AS 
						       (SELECT SPLIT(VALUES_COL,',') TGTCOLUMNS
				                  FROM `PROJECT_ID_31`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
				                 WHERE '`'||TARGET_PROJECT_ID||'`.'||TARGET_DATASET||'.'||TARGET_TABLE = TARGET_TABLE_PARAM
				               )
				        SELECT STRING_AGG(''||TGTCOLUMNS||' = '||'ODS.'||TGTCOLUMNS,',') CODN
				          FROM ARR
				         CROSS JOIN UNNEST(TGTCOLUMNS) AS TGTCOLUMNS
					   );

	  SELECT UPDATE_COLS;
				
	  /*SETTING INSERT_COLS RUNTIME */			
	  SET INSERT_COLS =(WITH
				         ARR AS 
						       (SELECT SPLIT(VALUES_COL,',') TGTCOLUMNS
				                  FROM `PROJECT_ID_31`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
				                 WHERE '`'||TARGET_PROJECT_ID||'`.'||TARGET_DATASET||'.'||TARGET_TABLE = TARGET_TABLE_PARAM
				               )
				        SELECT STRING_AGG('ODS.'||TGTCOLUMNS,',') CODN
				          FROM ARR
				         CROSS JOIN UNNEST(TGTCOLUMNS) AS TGTCOLUMNS
					   );

	  SELECT INSERT_COLS;

	  SET VAR_BUS_KEY_UPD =(WITH
				             ARR AS 
							       (SELECT SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
				                      FROM `PROJECT_ID_31`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
				                      WHERE '`'||TARGET_PROJECT_ID||'`.'||TARGET_DATASET||'.'||TARGET_TABLE = TARGET_TABLE_PARAM
				                   )
				            SELECT STRING_AGG('DIM.'||KEYCOLUMNS||' = '||'ODS.'||KEYCOLUMNS,' AND ') CODN
				              FROM ARR
				             CROSS JOIN UNNEST(KEYCOLUMNS) AS KEYCOLUMNS
						   );

	  SELECT VAR_BUS_KEY_UPD;

	  SET VAR_BUS_KEY_INS =(WITH
				             ARR AS 
							       (SELECT SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
				                      FROM `PROJECT_ID_31`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
				                     WHERE '`'||TARGET_PROJECT_ID||'`.'||TARGET_DATASET||'.'||TARGET_TABLE = TARGET_TABLE_PARAM
				                   )
				            SELECT STRING_AGG('ODS.'||KEYCOLUMNS,',') CODN
				              FROM ARR
				             CROSS JOIN UNNEST(KEYCOLUMNS) AS KEYCOLUMNS
						   );

	  SELECT VAR_BUS_KEY_INS;

	  SELECT VAR_ENDDTCOL ; 

	  SET VAR_MAX_LOAD_DATE_TIME = (SELECT MAX (DELTA_COLUMN_OFFSET) 
	                                  FROM `PROJECT_ID_31`.edw_data_migration_audit_tbls.JOB_AUDIT_LOG 
									 WHERE TABLE_NAME = TARGET_TABLE_PARAM
									   AND STATUS = 'SUCCESS'
								   );
	  SELECT VAR_MAX_LOAD_DATE_TIME;

	  IF VAR_MAX_LOAD_DATE_TIME IS NULL
	  THEN
		SET VAR_WHERE_CLAUSE = ' WHERE 1=1 ';
	  ELSE
		SET VAR_WHERE_CLAUSE = ' WHERE LOAD_DATE_TIME > '||'\''|| VAR_MAX_LOAD_DATE_TIME||'\'';
	  END IF;
	  
	  SELECT VAR_WHERE_CLAUSE;
	  
	  SET UPDATE_TO_TARGET= 
	              'UPDATE '
	              ||VAR_TARGET_PROJECT_ID
				  ||'.'
				  ||VAR_TARGET_DATASET
				  ||'.'
				  ||VAR_TARGET_TABLE
				  ||' DIM SET '
				  ||VAR_ENDDTCOL
				  ||' = DATE(ODS.'
				  ||VAR_STARTDTCOL
				  ||') - 1'
				  ||' FROM    '
				  ||VAR_SOURCE_PROJECT_ID
				  ||'.'
				  ||VAR_SOURCE_DATASET
				  ||'.'
				  ||VAR_TARGET_TABLE
				  ||'_TARGET ODS WHERE '
				  ||VAR_BUS_KEY_UPD
				  ||' AND (DIM.'
				  ||VAR_ENDDTCOL
				  ||'=CAST('
				  ||'\''
				  ||VAR_END_DATE
				  ||'\''
				  ||' AS DATE) OR DIM.' 
				  ||VAR_ENDDTCOL
                  ||' IS NULL)' 
				  ||'AND ODS.CHANGECODE IN ('
				  ||'\'U\''
				  ||');';

	  SELECT VAR_SCD_CATEGORY;	

	  IF VAR_SCD_CATEGORY = 'SCD2'
	  THEN
	    SELECT 'INSIDE SCD_TYPE2';
	    IF VAR_STARTDTCOL IS NULL OR VAR_ENDDTCOL IS NULL
	    THEN	
		select "LINE 472";
		  CALL `PROJECT_ID_30`.edw_data_migration_ops_tbls.SP_JOB_AUDIT_LOGGING (IFNULL(RUN_ID, 0), AUDIT_TABLE_NAME , AUDIT_TABLE_DATASET , 'FAIL' ,  JOB_ID, ' REQUIRED FIELDS FOR TYPE2 LOGIC NOT FOUND IN METADATA TABLE  ',RUN_DATE , SP_START_TIME, DATETIME_DIFF(CURRENT_DATETIME(),SP_START_TIME,SECOND), TARGET_TABLE_PARAM, NULL,CURRENT_DATETIME(),VAR_SCD_CATEGORY, EXECUTION_PARAMETERS, 'LOAD_DATE_TIME', VAR_LOG_LOAD_DATE_TIME);
		
		  SET SP_STATUS =1 ;
		  SELECT RECORD_COUNT, SP_STATUS;
		  RETURN;
	    ELSE	   		  
		  SET VAR_SQL_STR = 'SELECT COUNT(*) FROM '||VAR_TARGET_PROJECT_ID||'.'||VAR_TARGET_DATASET||'.'||VAR_TARGET_TABLE;
		  SELECT VAR_SQL_STR;
		  EXECUTE IMMEDIATE VAR_SQL_STR INTO VAR_TARGET_COUNT;
		  IF VAR_TARGET_COUNT = 0 
		  THEN
             SET VAR_SQL_STR = 
						'INSERT INTO '
						||VAR_TARGET_PROJECT_ID
						||'.'
						||VAR_TARGET_DATASET
						||'.'
						||VAR_TARGET_TABLE
						||' ('
						||VAR_BUS_KEY
						||','
						||VAR_VALUES_COL
						||','
						||VAR_STARTDTCOL
						||','
						||VAR_ENDDTCOL
