  EXECUTION_PARAMETERS = REPLACE(EXECUTION_PARAMETERS,'\'','\\\''); /* ASSIGING VALUES TO VARIABLES */

  (VAR_SCD_CATEGORY,
    VAR_SOURCE_DATASET,
    VAR_SOURCE_TABLE,

  VAR_LOG_LOAD_DATE_TIME_STR;

  VAR_LOG_LOAD_DATE_TIME_STR INTO VAR_LOG_LOAD_DATE_TIME;

  VAR_SQL_STR = 'SELECT COUNT(*) FROM '||VAR_SOURCE_DATASET||'.'||VAR_SOURCE_TABLE;

  VAR_SQL_STR;

  VAR_SQL_STR INTO VAR_SOURCE_COUNT;

  VAR_SOURCE_COUNT;

    JOB_ID,

    RUN_DATE,
    SP_START_TIME,
    DATETIME_DIFF(CURRENT_DATETIME(),SP_START_TIME,SECOND),

    NULL,
    CURRENT_DATETIME(),
    VAR_SCD_CATEGORY,

    'LOAD_DATE_TIME',
    VAR_LOG_LOAD_DATE_TIME);

  SP_STATUS =1 ;

  RECORD_COUNT,
  SP_STATUS; RETURN;
END IF
  ;
IF
  VAR_SCD_CATEGORY = 'FULL_LOAD' THEN
SELECT
  'INSIDE FULL LOAD';
SET
  VAR_SQL_STR = 'CREATE OR REPLACE TABLE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||'_TEMP AS SELECT * FROM ' ||VAR_SOURCE_DATASET ||'.' ||VAR_SOURCE_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
SET
  VAR_SQL_STR = 'ALTER TABLE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||' RENAME  TO ' ||VAR_TARGET_TABLE ||'_BACKUP_' ||FORMAT_DATE('%Y%m%d', RUN_DATE);
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
SET
  VAR_SQL_STR = 'ALTER TABLE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||'_TEMP' ||' RENAME  TO ' ||VAR_TARGET_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
SET
  VAR_SQL_STR = 'SELECT COUNT(*) FROM '||VAR_TARGET_DATASET||'.'||VAR_TARGET_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR INTO VAR_TARGET_COUNT;

  VAR_TARGET_COUNT;
IF
  VAR_SOURCE_COUNT = VAR_TARGET_COUNT THEN
SET
  VAR_SQL_STR = 'DROP TABLE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||'_BACKUP_' ||FORMAT_DATE('%Y%m%d', RUN_DATE);

  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
SET
  RECORD_COUNT = IFNULL(@@ROW_COUNT,0);
SET
  SP_STATUS = 0;
    ELSE
SET
  VAR_SQL_STR = 'DROP TABLE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
SET
  VAR_SQL_STR = 'ALTER TABLE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||'_BACKUP_' ||FORMAT_DATE('%Y%m%d', RUN_DATE) ||' RENAME TO ' ||VAR_TARGET_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
SET
  SP_STATUS = 1;
CALL
  `PROJECT_ID_30`.edw_data_migration_ops_tbls.SP_JOB_AUDIT_LOGGING (IFNULL(RUN_ID, 0),
    AUDIT_TABLE_NAME,
    AUDIT_TABLE_DATASET,
    'FAIL',
    JOB_ID,
    NULL,
    RUN_DATE,
    SP_START_TIME,
    NULL,
    VAR_TARGET_TABLE,
    NULL,
    NULL,
    VAR_SCD_CATEGORY,
    EXECUTION_PARAMETERS,
    'LOAD_DATE_TIME',
    VAR_LOG_LOAD_DATE_TIME);
END IF
  ;
SELECT
  RECORD_COUNT,
  SP_STATUS;
    ELSE
SET
  VAR_TARGET_COLUMNS_STR = 'SELECT STRING_AGG(COLUMN_NAME, '||'\''||','||'\''||' ORDER BY ORDINAL_POSITION) FROM '||VAR_TARGET_DATASET||'.INFORMATION_SCHEMA.COLUMNS WHERE '||'\''||'`'||'\''||'||UPPER(table_catalog) ||'||'\''||'`.'||'\''||'|| UPPER(TABLE_SCHEMA) ||'|| '\''||'.'||'\''|| '|| UPPER(TABLE_NAME) = '||'\''||UPPER(TARGET_TABLE_PARAM)||'\'';
SELECT
  VAR_TARGET_COLUMNS_STR;
EXECUTE IMMEDIATE
  VAR_TARGET_COLUMNS_STR INTO VAR_TARGET_COLUMN; /* SETTING MD5 COLUMN RUNTIME*/
SET
  VAR_MD5=(
  WITH
    INFO AS (
    SELECT
      COLUMN_NAME,
      CASE
        WHEN DATA_TYPE IN ('INT64', 'INTEGER', 'NUMERIC', 'BIGNUMERIC', 'BIGNUMERIC(38, 19)', 'FLOAT64') THEN 'IFNULL(CAST('||COLUMN_NAME||' AS STRING),\'-1\')'
        WHEN DATA_TYPE LIKE 'STRING%' THEN 'IFNULL('||COLUMN_NAME||',\'-1\')'
        WHEN DATA_TYPE='DATE' THEN 'IFNULL('||COLUMN_NAME||',\'1900-01-01\')'
        WHEN DATA_TYPE='TIME' THEN 'IFNULL('||COLUMN_NAME||',\'00:00:00\')'
        WHEN DATA_TYPE='DATETIME' THEN 'IFNULL('||COLUMN_NAME||',\'1900-01-01T00:00:00\')'
        WHEN DATA_TYPE='TIMESTAMP' THEN 'IFNULL('||COLUMN_NAME||',\'1900-01-01 00:00:00.000000 UTC\')'
        ELSE 'IFNULL('||COLUMN_NAME||',-1)'
    END
      AS ABI_COLS
    FROM
      `PROJECT_ID_30`.`region-us`.INFORMATION_SCHEMA.COLUMNS
    WHERE
      '`'||UPPER(table_catalog)||'`'|| '.'|| UPPER(TABLE_SCHEMA)||'.'||UPPER(TABLE_NAME) = UPPER(TARGET_TABLE_PARAM) ),
    TAB AS (
    SELECT
      SPLIT(VALUES_COL,',') AS TARGET_COLS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) ),
    TAB1 AS (
    SELECT
      TRIM(COLS) AS COLUMN_NAME
    FROM
      TAB
    CROSS JOIN
      UNNEST(TAB.TARGET_COLS) AS COLS )
  SELECT
    STRING_AGG(INFO.ABI_COLS,'||\'~\'||')
  FROM
    INFO
  INNER JOIN
    TAB1
  ON
    INFO.COLUMN_NAME = TAB1.COLUMN_NAME );
SELECT
  VAR_MD5; /* INSERTING RUNNING STATUS RECORD TO JOB AUDIT LOG TABLE */
CALL
  `PROJECT_ID_30`.edw_data_migration_ops_tbls.SP_JOB_AUDIT_LOGGING (IFNULL(RUN_ID, 0),
    AUDIT_TABLE_NAME,
    AUDIT_TABLE_DATASET,
    'RUNNING',
    JOB_ID,
    NULL,
    RUN_DATE,
    SP_START_TIME,
    NULL,
    VAR_TARGET_TABLE,
    NULL,
    NULL,
    VAR_SCD_CATEGORY,
    EXECUTION_PARAMETERS,
    'LOAD_DATE_TIME',
    VAR_LOG_LOAD_DATE_TIME); /* CHECKING IF ANY INVALID VALUES ARE PASSED AS PARAMETER AND MARKING THE SP FAILED IF YES */

  VAR_SOURCE_DATASET,
  VAR_SOURCE_TABLE,
  VAR_TARGET_DATASET,
  VAR_TARGET_TABLE,
  VAR_TARGET_COLUMN,
  VAR_BUS_KEY,
  VAR_VALUES_COL,
  VAR_STARTDTCOL;
IF
  VAR_SOURCE_DATASET IS NULL
  OR VAR_SOURCE_TABLE IS NULL
  OR VAR_TARGET_DATASET IS NULL
  OR VAR_TARGET_TABLE IS NULL
  OR VAR_TARGET_COLUMN IS NULL
  OR VAR_BUS_KEY IS NULL
  OR VAR_VALUES_COL IS NULL
  OR VAR_MD5 IS NULL THEN
CALL
  `PROJECT_ID_30`.edw_data_migration_ops_tbls.SP_JOB_AUDIT_LOGGING (IFNULL(RUN_ID, 0),
    AUDIT_TABLE_NAME,
    AUDIT_TABLE_DATASET,
    'FAIL',
    JOB_ID,
    'Invalid Metadata Entries ',
    RUN_DATE,
    SP_START_TIME,
    DATETIME_DIFF(CURRENT_DATETIME(),SP_START_TIME,SECOND),
    VAR_TARGET_TABLE,
    NULL,
    CURRENT_DATETIME(),
    VAR_SCD_CATEGORY,
    EXECUTION_PARAMETERS,
    'LOAD_DATE_TIME',
    VAR_LOG_LOAD_DATE_TIME);
SET
  SP_STATUS =1 ;
SELECT
  RECORD_COUNT,
  SP_STATUS; RETURN;
END IF
  ;
SET
  VAR_BUS_KEY_COL =(
  WITH
    ARR AS (
    SELECT
      SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) )
  SELECT
    STRING_AGG('MAIN.'||KEYCOLUMNS||' = '||'WRK.'||KEYCOLUMNS,' AND ') CODN
  FROM
    ARR
  CROSS JOIN
    UNNEST(KEYCOLUMNS) AS KEYCOLUMNS );
SELECT
  VAR_BUS_KEY_COL;
SET
  VAR_BUS_KEY_TGT_TMP =(
  WITH
    INFO AS (
    SELECT
      COLUMN_NAME,
      CASE
        WHEN DATA_TYPE IN ('STRING') THEN 'MAIN.'||COLUMN_NAME||'<>'||'\'-1\''
        ELSE 'MAIN.'||COLUMN_NAME||'<>'||'-1'
    END
      AS ABI_COLS
    FROM
      `PROJECT_ID_30`.`region-us`.INFORMATION_SCHEMA.COLUMNS
    WHERE
      '`'||UPPER(table_catalog)||'`'|| '.'|| UPPER(TABLE_SCHEMA)||'.'||UPPER(TABLE_NAME) = UPPER(TARGET_TABLE_PARAM) ),
    TAB AS (
    SELECT
      SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) ),
    TAB1 AS (
    SELECT
      TRIM(COLS) AS COLUMN_NAME
    FROM
      TAB
    CROSS JOIN
      UNNEST(TAB.KEYCOLUMNS) AS COLS )
  SELECT
    STRING_AGG(INFO.ABI_COLS,' AND ')
  FROM
    INFO
  INNER JOIN
    TAB1
  ON
    INFO.COLUMN_NAME = TAB1.COLUMN_NAME );
SELECT
  VAR_BUS_KEY_TGT_TMP;
SET
  VAR_VALUES_COL_INS =(
  WITH
    ARR AS (
    SELECT
      SPLIT(VALUES_COL,',') VALUES_COL1
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) )
  SELECT
    STRING_AGG('DIM.'||VALUES_COL1,',') CODN
  FROM
    ARR
  CROSS JOIN
    UNNEST(VALUES_COL1) AS VALUES_COL1 );
SELECT
  VAR_VALUES_COL_INS; /*SETTING UPDATE_COLS RUNTIME */
SET
  UPDATE_COLS =(
  WITH
    ARR AS (
    SELECT
      SPLIT(VALUES_COL,',') TGTCOLUMNS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) )
  SELECT
    STRING_AGG(''||TGTCOLUMNS||' = '||'ODS.'||TGTCOLUMNS,',') CODN
  FROM
    ARR
  CROSS JOIN
    UNNEST(TGTCOLUMNS) AS TGTCOLUMNS );
SELECT
  UPDATE_COLS; /*SETTING INSERT_COLS RUNTIME */
SET
  INSERT_COLS =(
  WITH
    ARR AS (
    SELECT
      SPLIT(VALUES_COL,',') TGTCOLUMNS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) )
  SELECT
    STRING_AGG('ODS.'||TGTCOLUMNS,',') CODN
  FROM
    ARR
  CROSS JOIN
    UNNEST(TGTCOLUMNS) AS TGTCOLUMNS );
SELECT
  INSERT_COLS;
SET
  VAR_BUS_KEY_UPD =(
  WITH
    ARR AS (
    SELECT
      SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) )
  SELECT
    STRING_AGG('DIM.'||KEYCOLUMNS||' = '||'ODS.'||KEYCOLUMNS,' AND ') CODN
  FROM
    ARR
  CROSS JOIN
    UNNEST(KEYCOLUMNS) AS KEYCOLUMNS );
SELECT
  VAR_BUS_KEY_UPD; /*Insert new records from ods(src) if not present for type4 logic*/
SET
  VAR_BUS_KEY_INS =(
  WITH
    ARR AS (
    SELECT
      SPLIT(BUS_KEY_COL,',') KEYCOLUMNS
    FROM
      `PROJECT_ID_30`.edw_data_migration_ops_tbls.T_METADATA_SCD_DATA_LOAD
    WHERE
      UPPER(TARGET_DATASET)||'.'||UPPER(TARGET_TABLE) = UPPER(TARGET_TABLE_PARAM) )
  SELECT
    STRING_AGG('ODS.'||KEYCOLUMNS,',') CODN
  FROM
    ARR
  CROSS JOIN
    UNNEST(KEYCOLUMNS) AS KEYCOLUMNS );
SELECT
  VAR_BUS_KEY_INS;
SELECT
  VAR_ENDDTCOL ;
SET
  VAR_MAX_LOAD_DATE_TIME = (
  SELECT
    MAX (DELTA_COLUMN_OFFSET)
  FROM
    `PROJECT_ID_31`.edw_data_migration_audit_tbls.JOB_AUDIT_LOG
  WHERE
    TABLE_NAME = VAR_TARGET_TABLE
    AND STATUS = 'SUCCESS' );
SELECT
  VAR_MAX_LOAD_DATE_TIME;
IF
  VAR_MAX_LOAD_DATE_TIME IS NULL THEN
SET
  VAR_WHERE_CLAUSE = ' WHERE 1=1 ';
    ELSE
SET
  VAR_WHERE_CLAUSE = ' WHERE LOAD_DATE_TIME > '||'\''|| VAR_MAX_LOAD_DATE_TIME||'\'';
END IF
  ;
SELECT
  VAR_WHERE_CLAUSE; /*UPDATE RECORD IN TARGET (in existing updated enddate to = run_date)*/ --
--SET UPDATE_TO_TARGET=CONCAT('UPDATE ',VAR_TARGET_DATASET,'.',VAR_TARGET_TABLE,' DIM SET ',VAR_ENDDTCOL,' = CAST(','\'',RUN_DATE,'\'',' AS DATE) FROM    ',VAR_TARGET_DATASET,'.',VAR_TARGET_TABLE,'_TARGET ODS WHERE ',VAR_BUS_KEY_UPD,' AND (DIM.',VAR_ENDDTCOL,'=CAST(','\'',VAR_END_DATE,'\'',' AS DATE) OR DIM.', VARENDDTCOL, ' IS NULL)', 'AND ODS.CHANGECODE IN (','\'U\'',',','\'D\'',');');
SET
  UPDATE_TO_TARGET= 'UPDATE ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||' DIM SET ' ||VAR_ENDDTCOL ||' = DATE(ODS.' ||VAR_STARTDTCOL ||') - 1' ||' FROM    ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||'_TARGET ODS WHERE ' ||VAR_BUS_KEY_UPD ||' AND (DIM.' ||VAR_ENDDTCOL ||'=CAST(' ||'\'' ||VAR_END_DATE ||'\'' ||' AS DATE) OR DIM.' ||VAR_ENDDTCOL ||' IS NULL)' ||'AND ODS.CHANGECODE IN (' ||'\'U\'' ||',' ||'\'D\'' ||');';
SELECT
  VAR_SCD_CATEGORY;
IF
  VAR_SCD_CATEGORY = 'SCD_TYPE2' THEN
SELECT
  'INSIDE SCD_TYPE2';
IF
  VAR_STARTDTCOL IS NULL
  OR VAR_ENDDTCOL IS NULL THEN
CALL
  `PROJECT_ID_30`.edw_data_migration_ops_tbls.SP_JOB_AUDIT_LOGGING (IFNULL(RUN_ID, 0),
    AUDIT_TABLE_NAME,
    AUDIT_TABLE_DATASET,
    'FAIL',
    JOB_ID,
    ' REQUIRED FIELDS FOR TYPE2 LOGIC NOT FOUND IN METADATA TABLE  ',
    RUN_DATE,
    SP_START_TIME,
    DATETIME_DIFF(CURRENT_DATETIME(),SP_START_TIME,SECOND),
    VAR_TARGET_TABLE,
    NULL,
    CURRENT_DATETIME(),
    VAR_SCD_CATEGORY,
    EXECUTION_PARAMETERS,
    'LOAD_DATE_TIME',
    VAR_LOG_LOAD_DATE_TIME);
SET
  SP_STATUS =1 ;
SELECT
  RECORD_COUNT,
  SP_STATUS; RETURN;
    ELSE
SET
  VAR_SQL_STR = 'SELECT COUNT(*) FROM '||VAR_TARGET_DATASET||'.'||VAR_TARGET_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR INTO VAR_TARGET_COUNT;
IF
  VAR_TARGET_COUNT = 0 THEN
SET
  VAR_SQL_STR = 'INSERT INTO ' ||VAR_TARGET_DATASET ||'.' ||VAR_TARGET_TABLE ||' (' ||VAR_BUS_KEY ||',' ||VAR_VALUES_COL ||',' ||VAR_STARTDTCOL ||',' ||VAR_ENDDTCOL ||')' ||' SELECT ' ||VAR_BUS_KEY ||',' ||VAR_VALUES_COL ||',' ||VAR_STARTDTCOL ||',' ||VAR_ENDDTCOL ||' FROM ' ||VAR_SOURCE_DATASET ||'.' ||VAR_SOURCE_TABLE;
SELECT
  VAR_SQL_STR;
EXECUTE IMMEDIATE
  VAR_SQL_STR;
    ELSE
SET
  VAR_LOAD_DATE_TIME_STR = '''SELECT ARRAY_AGG(LOAD_DATE_TIME ORDER BY LOAD_DATE_TIME) FROM (SELECT DISTINCT LOAD_DATE_TIME FROM ''' || VAR_SOURCE_DATASET || '''.''' || VAR_SOURCE_TABLE ||''' WHERE LOAD_DATE_TIME > (SELECT IFNULL(MAX(DELTA_COLUMN_OFFSET), (SELECT DATE(MIN (LOAD_DATE_TIME)) - INTERVAL 1 SECOND FROM ''' || VAR_SOURCE_DATASET || '''.''' || VAR_SOURCE_TABLE ||''')) FROM ''' || AUDIT_TABLE_DATASET || '''.''' ||AUDIT_TABLE_NAME || ''' WHERE TABLE_NAME = ''' ||'\'' || VAR_TARGET_TABLE ||'\'' || '''))''';
SELECT
  VAR_LOAD_DATE_TIME_STR;
EXECUTE IMMEDIATE
  VAR_LOAD_DATE_TIME_STR INTO VAR_LOAD_DATE_TIMES;
SELECT
  VAR_LOAD_DATE_TIMES;
SELECT
  ARRAY_LENGTH(VAR_LOAD_DATE_TIMES);
SELECT
  idx;
WHILE
  idx < ARRAY_LENGTH(VAR_LOAD_DATE_TIMES) DO
SET
  VAR_LOAD_DATE_TIME = VAR_LOAD_DATE_TIMES[
OFFSET
  (idx)];
SELECT
  VAR_LOAD_DATE_TIME;
SET
  VAR_WHERE_CLAUSE_SCD2 = ' WHERE LOAD_DATE_TIME = ' || '\'' ||VAR_LOAD_DATE_TIME ||'\'' ||' AND (' ||VAR_ENDDTCOL ||' IS NULL OR ' ||VAR_ENDDTCOL ||'  = CAST(' ||'\'' ||VAR_END_DATE ||'\'' ||' AS DATE))';
SET
  TEMP_FOR_TARGET = CONCAT('CREATE OR REPLACE TABLE ',VAR_TARGET_DATASET,'.',VAR_TARGET_TABLE,'_TARGET AS SELECT ',VAR_BUS_KEY,',', VAR_VALUES_COL,',',VAR_STARTDTCOL,',',VAR_ENDDTCOL,',','\'C\'',' AS CHANGECODE,','MD5(',VAR_MD5,') AS MD5',' FROM (','SELECT MAIN.* FROM ',VAR_TARGET_DATASET,'.',VAR_TARGET_TABLE,' MAIN  INNER JOIN ( SELECT DISTINCT ',VAR_BUS_KEY,' FROM ',VAR_SOURCE_DATASET,'.',VAR_SOURCE_TABLE,') WRK ON ',VAR_BUS_KEY_COL,' WHERE MAIN.',VAR_ENDDTCOL,' IS NULL OR MAIN.', VAR_ENDDTCOL, '  = CAST(','\'',VAR_END_DATE,'\'',' AS DATE));');
SELECT
  TEMP_FOR_TARGET;
EXECUTE IMMEDIATE
