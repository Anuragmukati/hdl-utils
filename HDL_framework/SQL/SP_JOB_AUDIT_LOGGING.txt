CREATE OR REPLACE PROCEDURE `PROJECT_ID_30`.edw_data_migration_ops_tbls.SP_JOB_AUDIT_LOGGING(RUN_ID INT64, LOG_TABLE_NAME STRING, LOG_TABLE_DATASET_NAME STRING, STATUS STRING, JOB_ID STRING, ERROR_MESSAGE STRING, RUN_DATE DATE, START_TIME DATETIME, EXECUTION_TIME INT64, TABLE_NAME STRING, ROW_AFFECTED INT64, END_TIME DATETIME, LOAD_TYPE STRING, EXECUTION_PARAMETERS STRING, DELTA_COLUMN STRING, DELTA_COLUMN_OFFSET DATETIME)
BEGIN
    DECLARE INSERT_QUERY, UPDATE_QUERY STRING;

    /* INSERTING A RUNNING STATUS RECORD IN AUDIT TABLE */

    IF STATUS = 'RUNNING' 
    THEN 
SELECT IFNULL (RUN_ID, 0);
        SET INSERT_QUERY = CONCAT('INSERT INTO ',LOG_TABLE_DATASET_NAME,'.',LOG_TABLE_NAME, '  (JOB_ID, STATUS, TABLE_NAME, LOAD_TYPE, EXECUTION_PARAMETERS, RUN_DATE, START_TIME, BATCH_ID, DELTA_COLUMN, DELTA_COLUMN_OFFSET, USER_ID) ', 
                                  ' VALUES( ', '\'', JOB_ID, '\'',' ,', '\'', STATUS, '\'',' ,', '\'', TABLE_NAME, '\'',' ,', '\'', LOAD_TYPE, '\'',' ,', '\'', EXECUTION_PARAMETERS, '\'',' ,', '\'', RUN_DATE, '\'',' ,', '\'', START_TIME , '\'',' ,', IFNULL (RUN_ID, 0), ',', '\'', DELTA_COLUMN, '\'', ',', '\'', DELTA_COLUMN_OFFSET, '\'', ',', '\'', SESSION_USER(), '\'', ')');
        SELECT INSERT_QUERY;
        EXECUTE IMMEDIATE  INSERT_QUERY;
    END IF;

    /* INSERTING A SUCCESS STATUS RECORD IN AUDIT TABLE */
    IF STATUS = 'SUCCESS' 
    THEN   
        SET INSERT_QUERY = CONCAT('INSERT INTO ',LOG_TABLE_DATASET_NAME,'.',LOG_TABLE_NAME, '  (JOB_ID, STATUS, TABLE_NAME, RUN_DATE, EXECUTION_TIME, START_TIME, END_TIME, LOAD_TYPE, EXECUTION_PARAMETERS, ROW_AFFECTED, BATCH_ID, DELTA_COLUMN, DELTA_COLUMN_OFFSET, USER_ID) ', 
                                        ' VALUES( ', '\'', JOB_ID, '\'',' ,', '\'', STATUS, '\'',' ,', '\'', TABLE_NAME, '\'',' ,', '\'', RUN_DATE , '\'',' ,', EXECUTION_TIME , ',', '\'', START_TIME , '\'',' ,', '\'', END_TIME , '\'',' ,', '\'', LOAD_TYPE, '\'',' ,', '\'', EXECUTION_PARAMETERS, '\'',' ,', ROW_AFFECTED , ' ,', IFNULL (RUN_ID, 0), ',', '\'',DELTA_COLUMN, '\'', ',', '\'', DELTA_COLUMN_OFFSET, '\'', ',', '\'', SESSION_USER(), '\'', ')');

        SELECT INSERT_QUERY;
       EXECUTE IMMEDIATE INSERT_QUERY;
 
    END IF;

    /* INSERTING A FAIL STATUS RECORD IN AUDIT TABLE */
    IF STATUS = 'FAIL'
     THEN
       SET INSERT_QUERY = CONCAT('INSERT INTO ',LOG_TABLE_DATASET_NAME,'.',LOG_TABLE_NAME, '  (JOB_ID, STATUS, TABLE_NAME, ERROR_MESSAGE, EXECUTION_TIME, START_TIME, END_TIME, LOAD_TYPE, EXECUTION_PARAMETERS, RUN_DATE, BATCH_ID, DELTA_COLUMN, DELTA_COLUMN_OFFSET, USER_ID) ', 
                                 ' VALUES( ', '\'' ,JOB_ID, '\'',' ,', '\'', STATUS, '\'',' ,', '\'', TABLE_NAME, '\'',' ,', '\'', ERROR_MESSAGE, '\'',' ,', EXECUTION_TIME , ',', '\'', START_TIME , '\'',' ,', '\'', END_TIME , '\'' ,' ,', '\'', LOAD_TYPE, '\'',' ,', '\'', EXECUTION_PARAMETERS, '\'',' ,', '\'', RUN_DATE, '\'', ' ,', IFNULL (RUN_ID, 0), ',', '\'', DELTA_COLUMN, '\'', ',', '\'', DELTA_COLUMN_OFFSET, '\'', ',', '\'', SESSION_USER(), '\'', ')');
    
        SELECT INSERT_QUERY;
        EXECUTE IMMEDIATE INSERT_QUERY;

    END IF;
	
	SET UPDATE_QUERY = 'UPDATE '
	                   ||LOG_TABLE_DATASET_NAME
                       ||'.'
                       ||LOG_TABLE_NAME
					   ||' SET DELTA_COLUMN_OFFSET = '
					   ||'\''
					   ||DELTA_COLUMN_OFFSET
					   ||'\''
					   ||' WHERE TABLE_NAME = '
					   ||'\''
					   ||TABLE_NAME 
					   ||'\''
					   ||' AND STATUS <> '
					   ||'\''
					   ||'SUCCESS'
					   ||'\'';
    
    SELECT UPDATE_QUERY;
    EXECUTE IMMEDIATE UPDATE_QUERY;	
					   
EXCEPTION WHEN ERROR THEN
	
    INSERT INTO PROJECT_ID_31.edw_data_migration_audit_tbls.JOB_AUDIT_LOG (JOB_ID, STATUS, TABLE_NAME, ERROR_MESSAGE, START_TIME, END_TIME, LOAD_TYPE, RUN_DATE, BATCH_ID, DELTA_COLUMN, DELTA_COLUMN_OFFSET, USER_ID) 
 VALUES(JOB_ID,'LOGGING FAILED', 'JOB_AUDIT_LOG', @@ERROR.MESSAGE, START_TIME , END_TIME , LOAD_TYPE, RUN_DATE,IFNULL (RUN_ID, 0), DELTA_COLUMN,DELTA_COLUMN_OFFSET,SESSION_USER());
    
END;
