#!/bin/bash
# ============================================================
# run_bq_ddl.sh
# Runs all TABLE DDLs first (5 in parallel), waits for all to
# complete, then runs all VIEW DDLs (5 in parallel).
# Captures SUCCESS / FAILURE with error details per file.
#
# Usage: bash run_bq_ddl.sh
# ============================================================

# ── Hardcoded config ─────────────────────────────────────────
GCP_PROJECT="PROJECT_ID_103"
TABLES_DIR="/path/to/Waves_DDL/Coasdo/Tables"
VIEWS_DIR="/path/to/Waves_DDL/Coasdo/Views"
BASE_DIR="/path/to/Waves_DDL/Coasdo"
PARALLEL=5
# ─────────────────────────────────────────────────────────────

TIMESTAMP=$(date +%Y%m%d_%H%M%S)
LOG_FILE="$BASE_DIR/ddl_run_${TIMESTAMP}.log"
RESULTS_CSV="$BASE_DIR/ddl_run_results_${TIMESTAMP}.csv"
TMP_DIR="$BASE_DIR/.ddl_tmp_${TIMESTAMP}"
mkdir -p "$TMP_DIR"

# ---------- log helper (thread-safe via file append) ----------
log() {
  echo "$1" | tee -a "$LOG_FILE"
}

# ---------- run a single DDL file (runs in background) --------
run_ddl() {
  local ddl_file="$1"
  local obj_type="$2"
  local obj_name
  obj_name=$(basename "$ddl_file" .sql)
  local tmp_out="$TMP_DIR/${obj_name}.result"

  ERROR_OUTPUT=$(bq query \
    --use_legacy_sql=false \
    --headless \
    --format=none \
    < "$ddl_file" 2>&1) && BQ_EXIT=0 || BQ_EXIT=$?

  if [[ $BQ_EXIT -eq 0 ]]; then
    echo "SUCCESS" > "$tmp_out"
    echo "$obj_name,$obj_type,$ddl_file,SUCCESS," >> "${tmp_out}.csv"
  else
    CLEAN_ERROR=$(echo "$ERROR_OUTPUT" | tr '\n' ' ' | tr ',' ';')
    echo "FAILED" > "$tmp_out"
    echo "$obj_name,$obj_type,$ddl_file,FAILED,\"$CLEAN_ERROR\"" >> "${tmp_out}.csv"
    echo "$ERROR_OUTPUT" >> "${tmp_out}.err"
  fi
}

# ---------- run all files in a dir, 5 at a time ---------------
run_parallel() {
  local dir="$1"
  local obj_type="$2"
  local pids=()
  local files=()

  mapfile -t files < <(find "$dir" -maxdepth 1 -name "*.sql" | sort)

  if [[ ${#files[@]} -eq 0 ]]; then
    log "WARNING: No .sql files found in $dir"
    return
  fi

  log "  Total files: ${#files[@]}, running $PARALLEL at a time"
  log ""

  local i=0
  for ddl_file in "${files[@]}"; do
    run_ddl "$ddl_file" "$obj_type" &
    pids+=($!)
    (( i++ )) || true

    # once we hit PARALLEL limit, wait for all current batch
    if (( i % PARALLEL == 0 )); then
      log "  Waiting for batch of $PARALLEL to complete..."
      for pid in "${pids[@]}"; do
        wait "$pid"
      done
      pids=()
      log "  Batch done."
    fi
  done

  # wait for any remaining jobs in the last partial batch
  for pid in "${pids[@]}"; do
    wait "$pid"
  done
}

# ---------- collect results and print per-file status ---------
collect_results() {
  local obj_type="$1"
  local success=0
  local failed=0

  for result_file in "$TMP_DIR"/*.result; do
    [[ -f "$result_file" ]] || continue
    local obj_name
    obj_name=$(basename "$result_file" .result)
    local status
    status=$(cat "$result_file")

    if [[ "$status" == "SUCCESS" ]]; then
      log "  [SUCCESS] $obj_name"
      (( success++ )) || true
    else
      log "  [FAILED]  $obj_name"
      if [[ -f "${result_file}.err" ]]; then
        log "  ERROR: $(cat "${result_file}.err" | tr '\n' ' ')"
      fi
      (( failed++ )) || true
    fi

    # append to final CSV
    if [[ -f "${result_file}.csv" ]]; then
      cat "${result_file}.csv" >> "$RESULTS_CSV"
    fi

    # clean up tmp files for this object
    rm -f "$result_file" "${result_file}.csv" "${result_file}.err"
  done

  echo "$success $failed"
}

# ══════════════════════════════════════════════════════════════
# MAIN
# ══════════════════════════════════════════════════════════════
echo "object_name,object_type,ddl_file,status,error" > "$RESULTS_CSV"

log "======================================================"
log " BQ DDL Run Started : $(date)"
log " Project            : $GCP_PROJECT"
log " Tables Dir         : $TABLES_DIR"
log " Views Dir          : $VIEWS_DIR"
log " Parallelism        : $PARALLEL"
log " Log File           : $LOG_FILE"
log " Results CSV        : $RESULTS_CSV"
log "======================================================"

TOTAL_SUCCESS=0
TOTAL_FAILED=0

# ── STEP 1: TABLES ────────────────────────────────────────────
log ""
log ">>> STEP 1: Creating TABLES (parallel=$PARALLEL) ..."
if [[ -d "$TABLES_DIR" ]]; then
  run_parallel "$TABLES_DIR" "TABLE"
  log ""
  log ">>> Collecting TABLE results ..."
  read -r t_success t_failed <<< "$(collect_results TABLE)"
  log "  Tables — Success: $t_success  |  Failed: $t_failed"
  (( TOTAL_SUCCESS += t_success )) || true
  (( TOTAL_FAILED  += t_failed  )) || true
else
  log "WARNING: Tables directory not found: $TABLES_DIR"
fi

log ""
log ">>> All TABLES done. Starting VIEWS ..."

# ── STEP 2: VIEWS ─────────────────────────────────────────────
log ""
log ">>> STEP 2: Creating VIEWS (parallel=$PARALLEL) ..."
if [[ -d "$VIEWS_DIR" ]]; then
  run_parallel "$VIEWS_DIR" "VIEW"
  log ""
  log ">>> Collecting VIEW results ..."
  read -r v_success v_failed <<< "$(collect_results VIEW)"
  log "  Views  — Success: $v_success  |  Failed: $v_failed"
  (( TOTAL_SUCCESS += v_success )) || true
  (( TOTAL_FAILED  += v_failed  )) || true
else
  log "WARNING: Views directory not found: $VIEWS_DIR"
fi

# ── SUMMARY ───────────────────────────────────────────────────
log ""
log "======================================================"
log " BQ DDL Run Completed : $(date)"
log "------------------------------------------------------"
log " Total Success : $TOTAL_SUCCESS"
log " Total Failed  : $TOTAL_FAILED"
log "======================================================"
log " Results CSV   : $RESULTS_CSV"
log "======================================================"

# cleanup temp dir
rm -rf "$TMP_DIR"
