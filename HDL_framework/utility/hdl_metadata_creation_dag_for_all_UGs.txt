VALID_APP_SUFFIXES = {
    "_PLE_Full":           "ple",
    "_PP_Full":            "prod",
    "_PP_Full_Daily":      "prod",
    "_PP_Full_Daily_R2":   "prod",
    "_PP_Incr_Daily":      "prod",

def get_project_from_app_name(app_name):
    app_upper = app_name.upper()
    matched_type = None
    for suffix, proj_type in VALID_APP_SUFFIXES.items():
        if app_upper.endswith(suffix.upper()):
            matched_type = proj_type
            break

    if matched_type is None:
        raise AirflowFailException(
            f"App_Name '{app_name}' does not match any valid pattern. "
            f"Valid suffixes: {', '.join(VALID_APP_SUFFIXES.keys())}"
        )

    prj        = app_name.split("_")[0].lower()
    config_key = f"bq_ple_project_{prj}" if matched_type == "ple" else f"bq_prod_project_{prj}"

    if config_key not in config:
        raise AirflowFailException(
            f"No configuration found for App_Name '{app_name}' (prj='{prj}'). "
            f"Missing key '{config_key}' in history_load_config_wave_hdl.json"
        )

    return config[config_key]

def parse_table_items(table_list):
    tables = []
    for item in table_list:
        item = str(item).strip()
        if '.' not in item:
            raise AirflowFailException(
                f"Invalid table entry '{item}'. Expected format: 'schema.table'"
            )
        parts = item.split('.', 1)
        tables.append((parts[0].strip(), parts[1].strip()))
    return tables

def update_bq_targets(**kwargs):
    TD_Table_list = kwargs['dag_run'].conf.get('TD_Table_list', [])
    BQ_Table_list = kwargs['dag_run'].conf.get('BQ_Table_list', [])
    App_Name      = kwargs['dag_run'].conf.get('App_Name', '')

    td_tables = parse_table_items(TD_Table_list)
    bq_tables = parse_table_items(BQ_Table_list)

    hook   = BigQueryHook(gcp_conn_id=hdl_source_connection_id, use_legacy_sql=False)
    conn   = hook.get_conn()
    cursor = conn.cursor()

    success_list = []
    skipped_list = []
    failed_list  = []

    for (td_db, td_tbl), (bq_db, bq_tbl) in zip(td_tables, bq_tables):
        if td_db.lower() == bq_db.lower() and td_tbl.lower() == bq_tbl.lower():
            skipped_list.append(f"{td_db}.{td_tbl}")
            continue

        bq_db_lower  = bq_db.lower()
        bq_tbl_lower = bq_tbl.lower()
        update_sql = (
            f"UPDATE `{HDL_project}.{Metadata_db}.{Metadata_tbl}` "
            f"SET "
            f"  tgt_table  = '{bq_tbl_lower}', "
            f"  tgt_schema = '{bq_db_lower}', "
            f"  table_uid  = '{bq_db}__{bq_tbl}', "
            f"  stg_table  = '{bq_db}__{bq_tbl}' "
            f"WHERE App_Name = '{App_Name}' "
            f"  AND LOWER(src_schema) = LOWER('{td_db}') "
            f"  AND LOWER(src_table)  = LOWER('{td_tbl}')"
        )
        try:
            cursor.execute(update_sql)
            success_list.append(f"{td_db}.{td_tbl} -> {bq_db}.{bq_tbl}  (rows: {cursor.rowcount})")
        except Exception as e:
            failed_list.append(f"{td_db}.{td_tbl}")
            print(f"  [FAIL] Update failed for {td_db}.{td_tbl}: {e}")
            raise AirflowFailException(f"BQ target update failed for {td_db}.{td_tbl}: {e}")

    print(f"\n{'='*60}")
    print(f"BQ TARGET UPDATE SUMMARY  |  App_Name: {App_Name}")
    print(f"{'='*60}")
    print(f"  Updated  : {len(success_list)}")
    print(f"  Skipped  : {len(skipped_list)}")
    print(f"  Failed   : {len(failed_list)}")

    if success_list:
        print(f"\n  [SUCCESS] BQ targets updated ({len(success_list)}):")
        for t in success_list:
            print(f"    + {t}")

    if skipped_list:
        print(f"\n  [SKIPPED] TD/BQ names identical, no update needed ({len(skipped_list)}):")
        for t in skipped_list:
            print(f"    ~ {t}")

    print(f"{'='*60}\n")
