def generate_query(**kwargs):
    
    #### Hitting Teradata and extracting data into GCS files
    ti = kwargs['ti']
    tbl_nm=kwargs['dag_run'].conf.get('table_name')
    col_name = kwargs['dag_run'].conf.get('column_name')

    
    #### Creating NOS SQL Query for Reading Data.
    sql_query = f"""
    SELECT MIN({col_name}) as min_val, MAX({col_name}) as max_val,cast(count(*) as bigint) as cnt FROM {tbl_nm};
    """

    ti.xcom_push(key='query', value=sql_query)
    # ti.xcom_push(key='gcs_path',value=gcs_path)
    # ti.xcom_push(key='gcs_bucket',value=gcs_bucket)
    print("Generated Query:", sql_query)


def clean_error_msg(error_msg):
    msg=str(error_msg)
    msg=re.sub(r"[\n\r\t]+"," ",msg)
    msg=re.sub(r"[\"']","",msg)
    msg=re.sub(r"\s{2,}"," ",msg)
    return msg.strip()


def fail_task(context):
    task_instance=context.get('task_instance')
    exception=context.get('exception')
    error=clean_error_msg(str(task_instance)+" "+str(exception))
    print("DAG execution failed!",error)

def run_td_sql(sql):
    conn=BaseHook.get_connection(functional_id_used)
    try:
        #password = access_secret_version("td_pwd")
        
        
        connection =teradatasql.connect(
            host=conn.host,
            user=conn.login,
            password=conn.password
        )

        cursor = connection.cursor()
        
        print("sql: ",sql)
        cursor.execute(sql)
        count = cursor.fetchone()
        cursor.close()
        connection.close()
        return count

    except Exception as e:
        print(f"**** Error while getting Min,Max for table: {str(e)}")
        return -1
