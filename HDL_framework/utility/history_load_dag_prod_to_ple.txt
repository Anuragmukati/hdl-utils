from airflow import DAG
from airflow.operators.python import PythonOperator
from airflow.providers.google.cloud.hooks.bigquery import BigQueryHook
from airflow.exceptions import AirflowFailException
from google.cloud.exceptions import NotFound
import json
from datetime import datetime
import sys

BASE_DIR = "/home/airflow/gcs/dags/PROJECT_ID_25/"
sys.path.append(f"{BASE_DIR}/")


def read_config():
    with open('/home/airflow/gcs/dags/PROJECT_ID_25/history_load_config_wave_hdl.json', 'r') as f:
        confid_details = json.load(f)
    return confid_details

config = read_config()

hdl_source_connection_id = config["hdl_source_connection_id"]
instances_cfg_tbl = f"{config['hdl_source_project']}.{config['Metadata_Dataset']}.{config.get('instances_config_tbl','hdl_project_instances_config')}"


def check_table_exists(BQ_Table_list, GCP_CONN_DO_ID, GCP_CONN_DO_ID_PLE, bq_prod_project, bq_ple_project):
    hook = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID_PLE, use_legacy_sql=False)
    client = hook.get_client()
    tbl_fl_nm = f'{bq_ple_project}.{BQ_Table_list}'
    tbl_fl_nm_prd = f'{bq_prod_project}.{BQ_Table_list}'
    prd_status = ''
    ple_status = ''
    try:
        client.get_table(tbl_fl_nm)
        print(f"Table is exists:{tbl_fl_nm}")
        ple_status = 'True'
    except NotFound:
        print(f"Table is not exists:{tbl_fl_nm}")
        ple_status = 'False'
    hook = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID, use_legacy_sql=False)
    client = hook.get_client()
    try:
        client.get_table(tbl_fl_nm_prd)
        print(f"Table is exists:{tbl_fl_nm_prd}")
        prd_status = 'True'
    except NotFound:
        print(f"Table is not exists:{tbl_fl_nm_prd}")
        prd_status = 'False'

    return ple_status, prd_status

def check_ddl_match(BQ_Table_list, GCP_CONN_DO_ID, GCP_CONN_DO_ID_PLE, bq_prod_project, bq_ple_project):
    prd_client = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID, use_legacy_sql=False).get_client()
    ple_client = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID_PLE, use_legacy_sql=False).get_client()

    prd_schema = prd_client.get_table(f'{bq_prod_project}.{BQ_Table_list}').schema
    ple_schema = ple_client.get_table(f'{bq_ple_project}.{BQ_Table_list}').schema

    def schema_to_list(schema):
        return [(f.name.lower(), f.field_type.upper(), f.mode.upper()) for f in schema]

    prd_fields = schema_to_list(prd_schema)
    ple_fields = schema_to_list(ple_schema)

    if prd_fields == ple_fields:
        print("DDL check passed: PROD and PLE schemas match.")
        return True

    prd_set = set(prd_fields)
    ple_set = set(ple_fields)
    only_in_prd = prd_set - ple_set
    only_in_ple = ple_set - prd_set
    print("#######################################")
    print("DDL MISMATCH between PROD and PLE!")
    if only_in_prd:
        print(f"Columns/types in PROD but not in PLE: {sorted(only_in_prd)}")
    if only_in_ple:
        print(f"Columns/types in PLE but not in PROD: {sorted(only_in_ple)}")
    print("#######################################")
    return False

def prod_to_ple(**kwargs):
    Table_Name_list = kwargs['dag_run'].conf.get('Table_Name', [])
    prj = str(kwargs['dag_run'].conf.get('Project', '')).lower()

    if isinstance(Table_Name_list, str):
        Table_Name_list = [Table_Name_list]

    inst_hook = BigQueryHook(gcp_conn_id=hdl_source_connection_id, use_legacy_sql=False)
    inst_cursor = inst_hook.get_conn().cursor()
    inst_cursor.execute(f"SELECT prj FROM `{instances_cfg_tbl}`")
    valid_projects = [row[0] for row in inst_cursor.fetchall()]
    print(f"Valid projects loaded from BQ config: {valid_projects}")

    if prj not in valid_projects:
        print("###############################")
        print("##### Project is Wrong ########")
        print("###############################")
        raise AirflowFailException(f"Invalid Project: '{prj}'. Valid options: {valid_projects}")

    GCP_CONN_DO_ID = config[f"GCP_CONN_DO_ID_{prj}"]
    GCP_CONN_DO_ID_PLE = config[f"GCP_CONN_DO_ID_PLE_{prj}"]
    bq_prod_project = config[f"bq_prod_project_{prj}"]
    bq_ple_project = config[f"bq_ple_project_{prj}"]

    hook = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID, use_legacy_sql=False)
    conn = hook.get_conn()
    cursor = conn.cursor()

    for Table_Name in Table_Name_list:
        print(f"Processing table: {Table_Name}")
        ple, prd = check_table_exists(Table_Name, GCP_CONN_DO_ID, GCP_CONN_DO_ID_PLE, bq_prod_project, bq_ple_project)

        if ple == 'True' and prd == 'True':
            ddl_match = check_ddl_match(Table_Name, GCP_CONN_DO_ID, GCP_CONN_DO_ID_PLE, bq_prod_project, bq_ple_project)
            if not ddl_match:
                print(f"Skipping {Table_Name} due to DDL mismatch. Please align the schemas before copying.")
                continue
            sql = f"BEGIN TRANSACTION; TRUNCATE TABLE {bq_ple_project}.{Table_Name}; INSERT INTO {bq_ple_project}.{Table_Name} SELECT * FROM {bq_prod_project}.{Table_Name}; COMMIT TRANSACTION;"
            try:
                print(sql)
                cursor.execute(sql)
                print(f"Successfully copied {Table_Name} from PROD to PLE.")
            except Exception as e:
                print(f'Error occured: {e}')
                print(f'PROD to PLE got failed for table: {Table_Name}')
                raise
        else:
            if ple != 'True' and prd != 'True':
                print("#######################################")
                print("Table not created in PLE and Prod Both", Table_Name)
                print("#######################################")
            elif ple != 'True':
                print("#######################################")
                print("Table not created in ple", Table_Name)
                print("#######################################")
            elif prd != 'True':
                print("#######################################")
                print("Table not created in prod", Table_Name)
                print("#######################################")

default_args = {
    'owner': 'airflow',
    'depends_on_past': False,
    'start_date': datetime(2025, 4, 9),
    'retries': 0
}

dag = DAG(
    dag_id='history_load_dag_prod_to_ple',
    default_args=default_args,
    description='DAG to generate NOS SQL query for Teradata',
    schedule_interval=None,
    max_active_runs=5,
    catchup=False,
    params={
        "Table_Name": ["db.table1", "db.table2"],
        "Project": "bglodo/bgtsdo/bhubdo/bopsdo/bsopdo/bsyido/factdo/fcredo/ffrado/frvado/vcgfdo/zdwvdo/zdwvbizdo"
    }
)

bq_prod_to_bq_ple_load = PythonOperator(
    task_id='bq_prod_to_bq_ple_load',
    python_callable=prod_to_ple,
    provide_context=True,
    dag=dag
)

bq_prod_to_bq_ple_load
