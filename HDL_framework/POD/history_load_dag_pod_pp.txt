def extract_metadata(table_id):
    sql = f"""select is_table_active,table_uid,src_schema,src_table,stg_schema,stg_table,tgt_schema,tgt_table,load_type,delta_column,delta_column_type,delta_column_format,delta_interval,app_name,src_table_size,src_table_row_count,src_table_type,bus_key_column,change_value_column,start_date_column,end_date_column,stg_load_enabled,plican_validation_enabled,select_query,load_functionality,select_query_tpt from `{HDL_project}.{Metadata_db}.{Metadata_tbl}`
    where  is_table_active=True and concat(tgt_schema,'.',tgt_table)='{table_id}' limit 1;"""
    print(f"**** Metadata select query: \n {sql}")

    hook = BigQueryHook(gcp_conn_id=hdl_source_connection_id, use_legacy_sql=False)
    conn = hook.get_conn()
    cursor = conn.cursor()
    cursor.execute(sql)
    row = cursor.fetchone()

    print(f"**** Total metadata table rows fetched: {len(row)}")
    table_info = {}
    table_info['tgt_schema'] = row[6]
    table_info['tgt_table'] = row[7]
    table_info['delta_column'] = row[9]
    return table_info

    if last_val:
        return 0
    elif last_val is None:
        print(f"***** Max value not found for delta column:{metadata.get('delta_column')} , table:{table_id}")
        return 1
    else:
        -1

def queue_counter(queue_counts, table_name,non_of_instacnes):
    qnum = queue_counts[table_name]
    queue_counts[table_name] = qnum+1 if qnum < non_of_instacnes else 1
    return qnum,queue_counts


def get_delta_max_value(schema,table_name):
    query=f"""SELECT delta_column_max_value FROM `{HDL_project}`.{Audit_db}.{Audit_tbl}  WHERE concat(tgt_schema,'.',tgt_table) ='{schema}.{table_name}' and app_name='NOS_First'  AND Job_STATUS = 'SUCCESS' order by end_time desc"""

    print(f"**** get max value query: {query}")
    hook = BigQueryHook(gcp_conn_id=hdl_source_connection_id, use_legacy_sql=False)
    conn = hook.get_conn()
    cursor = conn.cursor()
    try:
        cursor.execute(query)
        rows = cursor.fetchall()
        return rows[0][0] if rows and rows[0][0] is not None else None
    except Exception as e:
        print(f"Error while fetching running tasks")
        print(f"Error: ",str(e))
        print(f"Traceback: ",traceback.format_exc())
        return -1
