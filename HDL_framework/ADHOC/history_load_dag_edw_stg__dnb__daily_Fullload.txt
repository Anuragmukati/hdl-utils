def send_email_report(Status, **kwargs):
    dag_run_id = kwargs['dag'].dag_id
    subject = f"HDL DAG Execution Report: {dag_run_id}"
    recipient_emails = [
        "user@example.com"
    ]
    sender_email = "user@example.com"

    html_content = f"""
    <html>
    <body>
    <h1>DAG Execution Completed</h1>
    <p>The DAG <b>{dag_run_id}</b> has {Status}.</p>
    </body>
    </html>
    """

    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = sender_email
    msg["To"] = ", ".join(recipient_emails)
    msg.attach(MIMEText(html_content, "html"))

    smtp_server = "smtp.example.com"
    smtp_port = 25
    with smtplib.SMTP(smtp_server, smtp_port) as server:
        server.ehlo()
        server.starttls()
        server.sendmail(sender_email, recipient_emails, msg.as_string())

    print("Email sent successfully.")


def get_td_tbl_count(td_schema, td_table, cnt_tbl_sql=None):

        print("**** Getting td row count")
        connection = teradatasql.connect(

        if not cnt_tbl_sql:

def process_full_load_in_hash_chunks(**kwargs):
    ti = kwargs['ti']
    stg_db = ti.xcom_pull(task_ids='fetch_metadata', key='stg_db')
    stg_tbl = ti.xcom_pull(task_ids='fetch_metadata', key='stg_tbl')
    fetch_query = ti.xcom_pull(task_ids='fetch_metadata', key='fetch_query')
    dt = kwargs['execution_date'].strftime('%Y%m%dT%H%M%S')

    bq_hook = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID, use_legacy_sql=False)
    bq_client = bq_hook.get_client(project_id=bq_prod_project)

    gcs_hook = BigQueryHook(gcp_conn_id=GCS_CONN_DO_ID, use_legacy_sql=False)
    storage_client = storage.Client(project=bq_prod_project, credentials=gcs_hook.get_client()._credentials)

    table_id = f"{bq_prod_project}.{stg_db}.{stg_tbl}"
    conn = BaseHook.get_connection(functional_id_used)
    td_connection = teradatasql.connect(host=conn.host, user=conn.login, password=conn.password)
    td_cursor = td_connection.cursor()

    any_chunk_loaded = False
    try:
        for chunk_id in range(TOTAL_HASH_CHUNKS):
            gcs_path = f"nos/input/opsstg_tbls_{stg_tbl}/{dt}/chunk_{chunk_id}"
            chunk_query = f"SELECT * FROM ({fetch_query}) x WHERE MOD(HASHBUCKET(HASHROW({HASH_COLUMN})), {TOTAL_HASH_CHUNKS}) = {chunk_id}"
            nos_sql = f"SELECT * FROM WRITE_NOS (ON ( {chunk_query} ) USING LOCATION('/gs/storage-ENDPOINT.p.googleapis.com/{gcs_bucket}/{gcs_path}/') AUTHORIZATION({TD_Auth_used}) STOREDAS('parquet') MAXOBJECTSIZE('16MB') COMPRESSION('SNAPPY')) AS d;"
            
            print(f"Running chunk {chunk_id + 1}")
            td_cursor.execute(nos_sql)

            bucket = storage_client.bucket(gcs_bucket)
            if list(bucket.list_blobs(prefix=gcs_path)):
                job_config = bigquery.LoadJobConfig(
                    source_format=bigquery.SourceFormat.PARQUET,
                    write_disposition=bigquery.WriteDisposition.WRITE_APPEND if any_chunk_loaded else bigquery.WriteDisposition.WRITE_TRUNCATE
                )
                bq_client.load_table_from_uri(f"gs://{gcs_bucket}/{gcs_path}/*.parquet", table_id, job_config=job_config).result()
                any_chunk_loaded = True

        ti.xcom_push(key='delta_tgt_cnt', value=get_bq_tbl_count(stg_db, stg_tbl))
    finally:
        td_connection.close()

def clean_error_msg(error_msg):
    msg = str(error_msg)
    msg = re.sub(r"[\n\r\t]+", " ", msg)
    msg = re.sub(r"[\"']", "", msg)
    msg = re.sub(r"\s{2,}", " ", msg)
    return msg.strip()


def fail_task(context):
    task_instance = context.get('task_instance')
    exception = context.get('exception')
    error = clean_error_msg(str(task_instance) + " " + str(exception)) + "|Hash chunk load"
    updt_sql("Dag_Failed", error_message=error, **context)
    send_email_report("Dag Got Failed", **context)
