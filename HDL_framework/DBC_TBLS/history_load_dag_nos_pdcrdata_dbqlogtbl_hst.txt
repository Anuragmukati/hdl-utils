def fetch_metadata(**kwargs):
    where_condition=kwargs['dag_run'].conf.get('where_condition','')
    fetch_query=f"select * from pdcrdata.dbqlogtbl_hst {where_condition} "
    db_nm="edw_data_migration_ops_tbls"
    tbl_nm="pdcrdata_dbqlogtbl_hst"
    
    ti = kwargs['ti']
    ti.xcom_push(key='fetch_query', value=fetch_query)
    ti.xcom_push(key='tbl_nm',value=tbl_nm)
    ti.xcom_push(key='db_nm',value=db_nm)

def generate_nos_query(**kwargs):
    ti = kwargs['ti']
    db_nm=ti.xcom_pull(task_ids='fetch_metadata', key='db_nm') 
    tbl_nm=ti.xcom_pull(task_ids='fetch_metadata', key='tbl_nm')
    fetch_query=ti.xcom_pull(task_ids='fetch_metadata', key='fetch_query') 
    bucket = "BUCKET_NAME_02"
    dt=kwargs['execution_date'].strftime('%Y%m%dT%H%M%S')
    gcs_path = "nos/input/"+str(db_nm)+'_'+str(tbl_nm)+'/'+str(dt)
    #bucket = "BUCKET_NAME_02"
    #gcs_path = "nos/"+str(db_nm)+'_'+str(tbl_nm)
    sql_query = f"""
    SELECT * FROM WRITE_NOS (
        ON ( {fetch_query} )
        USING
        LOCATION('/gs/storage.googleapis.com/{bucket}/{gcs_path}/')
        AUTHORIZATION(<Teradata NOS authorization object with write access to the project GCS bucket>)
        STOREDAS('parquet')
        MAXOBJECTSIZE('16MB')
        COMPRESSION('SNAPPY')
    ) AS d;
    """

    ti.xcom_push(key='nos_query', value=sql_query)
    ti.xcom_push(key='gcs_path',value=gcs_path)
    ti.xcom_push(key='gcs_bucket',value=bucket)
    

    print("Generated Query:", sql_query)
    #updt_sql('INPROGRESS',db_nm,tbl_nm)

def fail_task(context):
    ti=context['ti']
    db_nm=ti.xcom_pull(task_ids='fetch_metadata', key='db_nm')
    tbl_nm=ti.xcom_pull(task_ids='fetch_metadata', key='tbl_nm')
    #updt_sql('DAG_FAILED',db_nm,tbl_nm)
    raise AirflowFailException("DAG execution failed!")


default_args = {
    'owner': 'airflow',
    'depends_on_past': False,
    'start_date': datetime(2025, 5, 30),
    'retries': 0
}

dag = DAG(
    dag_id='history_load_dag_nos_pdcrdata_dbqlogtbl_hst1',
    default_args=default_args,
    description='DAG to generate NOS SQL query for Teradata',
    schedule_interval=None,
    max_active_runs=1,
    catchup=False
)




    
get_metadata = PythonOperator(
    task_id='fetch_metadata',
    python_callable=fetch_metadata,
    provide_context=True,  # Ensures kwargs include `dag_run`
    dag=dag,
    on_failure_callback=fail_task
)   


generate_query_task = PythonOperator(
    task_id='generate_nos_query',
    python_callable=generate_nos_query,
    provide_context=True,  # Ensures kwargs include `dag_run`
    dag=dag,
    on_failure_callback=fail_task
)

nos_to_gcs_load = TeradataOperator(
    task_id='nos_to_gcs_load',
    conn_id="P2_FUNCTION_ID_1",
    sql="{{ ti.xcom_pull(task_ids='generate_nos_query', key='nos_query') }}",
    retries=0,
    dag=dag,
    on_failure_callback=fail_task
)

gcs_to_bq_load = GCSToBigQueryOperator(
    task_id="gcs_to_bq_load",
    dag=dag,
    bucket="BUCKET_NAME_02",
    write_disposition="WRITE_APPEND",
    source_format="PARQUET",
    allow_jagged_rows=True,
    allow_quoted_newlines=True,
    quote_character="\"",
    gcp_conn_id=GCP_CONN_DO_ID,
    source_objects=["{{ti.xcom_pull(task_ids='generate_nos_query', key='gcs_path') +'/*.parquet'}}"],
    destination_project_dataset_table="{{params.bq_project}}.{{ ti.xcom_pull(task_ids='fetch_metadata', key='db_nm') }}.{{ ti.xcom_pull(task_ids='fetch_metadata', key='tbl_nm') }}",
    on_failure_callback=fail_task,
    params={'bq_project':bq_target_project}
)
