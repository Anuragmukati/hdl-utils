def format_td_timestamp(ts_str):
    if not ts_str:
        return None
    return ts_str.replace('T', ' ').split('.')[0]


def format_td_time(tm_str):
    if not tm_str:
        return None
    return tm_str.split('.')[0]


def task_status_callback(context):
    ti = context['task_instance']
    map_index  = ti.map_index
    all_metadata = ti.xcom_pull(task_ids='bulk_audit_init')
    if not all_metadata or map_index >= len(all_metadata):
        return
    conf = all_metadata[map_index]
    start_time_est = (
        pendulum.instance(ti.start_date).in_timezone('America/New_York').format('YYYY-MM-DD HH:mm:ss')
        if ti.start_date else None
    )
    end_time_est = (
        pendulum.instance(ti.end_date).in_timezone('America/New_York').format('YYYY-MM-DD HH:mm:ss')
        if ti.end_date else pendulum.now('America/New_York').format('YYYY-MM-DD HH:mm:ss')
    )
    state  = ti.state
    status = "SUCCESS" if state == "success" else "FAILED"
    bq_count, td_count = None, None

    if state == "success":
        try:
            counts    = fetch_counts(conf.copy())
            bq_count  = counts.get('bq_count')
            td_count  = counts.get('td_count')
            if bq_count is not None and td_count is not None:
                if int(bq_count) != int(td_count):
                    status = "COUNT_MISMATCH"
            else:
                status = "COUNT_FETCH_FAILED"
        except Exception as e:
            logger.error(f"Count fetch failed for {conf.get('bq_table_name')}: {e}")
            status = "COUNT_FETCH_FAILED"

    result_payload = {
        **conf,
        "status": status,
        "bq_count": bq_count,
        "td_count": td_count,
        "start_time": start_time_est,
        "end_time": end_time_est
    }
    ti.xcom_push(key="final_audit_record", value=result_payload)


def fetch_counts(record: dict):
    project_nm = record.get('project_name', BQ_PROJECT)
    bq_hook    = BigQueryHook(gcp_conn_id=BQ_CONN_ID)
    try:
        bq_query  = f"SELECT CAST(COUNT(*) AS BIGINT) AS cnt FROM `{project_nm}.{record['bq_table_name']}`"
        df        = bq_hook.get_pandas_df(sql=bq_query, dialect="standard", location="US")
        bq_count  = int(df['cnt'][0])
    except Exception as e:
        logger.warning("Failed to fetch BQ count for %s: %s", record['bq_table_name'], e)
        bq_count = None

    try:
        conn_info = BaseHook.get_connection(TD_CONN_ID)
        with teradatasql.connect(host=conn_info.host, user=conn_info.login, password=conn_info.password) as td_conn:
            td_cursor = td_conn.cursor()
            td_cursor.execute(f"SELECT CAST(COUNT(*) AS BIGINT) FROM {record['td_table_name']}")
            td_count = int(td_cursor.fetchone()[0])
            td_cursor.close()
    except Exception as e:
        logger.warning("Failed to fetch TD count for %s: %s", record['td_table_name'], e)
        td_count = None

    record.update({
        'bq_count':    bq_count,
        'td_count':    td_count,
        'data_size_gb': record.get('bq_table_logical_size_gb', 'NA')
    })
    logger.info("Fetched counts for %s — BQ: %s | TD: %s | Size: %s GB",
                record['bq_table_name'], bq_count, td_count, record['data_size_gb'])
    return record


with DAG(
    dag_id=DAG_ID,
    start_date=datetime(2026, 3, 9, tzinfo=local_tz),
    schedule_interval=None,
    max_active_runs=10,
    max_active_tasks=50,
    catchup=True,
    default_args=default_args,
    description="Manual trigger: writeback DAGs with BQ & Teradata counts and audit tracking",
    params={"bq_table_names": ""}
) as dag:

    start = EmptyOperator(task_id="start")
    end   = EmptyOperator(task_id="end", trigger_rule=TriggerRule.ALL_DONE)

    @task
    def fetch_metadata(**context):
        conf           = context.get("dag_run").conf or {}
        hook           = BigQueryHook(gcp_conn_id=BQ_CONN_ID)
        dag_name_filter= conf.get("bq_table_names")

        if not dag_name_filter:
            raise ValueError("bq_table_names param is required for manual trigger.")

        query = f"""
SELECT
    m.*,
    ROUND(s.total_logical_bytes / POW(1024, 3), 2) AS bq_table_logical_size_gb,
    COALESCE(CAST(s.storage_last_modified_time AS STRING), '') AS storage_last_modified_time
FROM `{BQ_PROJECT}.region-us.INFORMATION_SCHEMA.TABLE_STORAGE` s
INNER JOIN `{BQ_METADATA_TABLE}` m
    ON (s.table_schema || '.' || s.table_name) = m.bq_table_name
WHERE
    m.is_active = TRUE
    AND m.load_instances BETWEEN {LOAD_INSTANCES_MIN} AND {LOAD_INSTANCES_MAX}
    AND m.app_name = '{project}'
    AND m.bq_table_name IN ({dag_name_filter})
"""
        logger.info("Fetching manual metadata for project: %s", project)
        print(query)

        df = hook.get_pandas_df(sql=query, dialect="standard", location="US")

        for col in df.columns:
            if pd.api.types.is_datetime64_any_dtype(df[col]):
                df[col] = df[col].apply(lambda x: x.strftime('%Y-%m-%d %H:%M:%S') if pd.notna(x) else None)

        df = df.replace({pd.NaT: None, pd.NA: None}).where(pd.notnull(df), None)

        records_cleaned = df.drop_duplicates().to_dict(orient='records')

        if not records_cleaned:
            logger.warning("No matching metadata found for: %s", dag_name_filter)

        return records_cleaned

    @task
    def bulk_audit_init(metadata_list: list, **context):
        ti = context['task_instance']
        if not metadata_list:
            return
        hook           = BigQueryHook(gcp_conn_id=BQ_CONN_ID)
        exec_hour      = context['execution_date'].strftime('%H:00')
        start_time_est = (
            pendulum.instance(ti.start_date).in_timezone('America/New_York').format('YYYY-MM-DD HH:mm:ss')
            if ti.start_date else None
        )

        values_list = []
        for record in metadata_list:
            values_list.append(f"""(
                '{record['bq_table_name']}', '{record['td_table_name']}', '{exec_hour}', 'RUNNING',
                '{record.get('bq_table_logical_size_gb', '')}', '{record.get('function_id', '')}',
                '{record.get('writeback_composer_instance', '')}', '{record['writeback_dag_name']}',
                '{record.get('load_type', '')}', '{record.get('load_functionality', '')}',
                CURRENT_TIMESTAMP(), '{start_time_est}', '{record['storage_last_modified_time']}', 'manual_triggered'
            )""")

        insert_query = f"""
            INSERT INTO `{BQ_AUDIT_TABLE}`
            (bq_table_name, td_table_name, execution_hour, writeback_status, data_size_gb,
             functional_id, composer_instance, writeback_dag_name, load_type, load_functionality,
             audit_ts, start_time, table_last_modified_when_wb, trigger_type)
            VALUES {", ".join(values_list)}
        """
        hook.run_query(sql=insert_query, use_legacy_sql=False)
        return metadata_list

    metadata_records = fetch_metadata()
    init_audit_data  = bulk_audit_init(metadata_records)

    trigger_worker = TriggerDagRunOperator.partial(
        task_id="trigger_dag",
        wait_for_completion=True,
        deferrable=True,
        poke_interval=60,
        on_success_callback=task_status_callback,
        on_failure_callback=task_status_callback,
        execution_timeout=timedelta(hours=6),
        retries=0
    ).expand_kwargs(
        init_audit_data.map(lambda x: {
            "trigger_dag_id": x["writeback_dag_name"],
            "conf": x,
            "task_id": f"trigger_{x['bq_table_name'].replace('.', '_')}"
        })
    )

    @task(trigger_rule=TriggerRule.ALL_DONE)
    def bulk_audit_finalize(**context):
        ti          = context['ti']
        all_results = ti.xcom_pull(task_ids='trigger_dag', key='final_audit_record', map_indexes=None)
        if not all_results:
            logger.warning("No audit records found in XCom.")
            return

        valid_results = [r for r in all_results if r is not None]
        values_list   = []
        for res in valid_results:
            bq_cnt     = f"'{res['bq_count']}'"  if res['bq_count'] is not None else "NULL"
            td_cnt     = f"'{res['td_count']}'"  if res['td_count'] is not None else "NULL"
            end_time   = f"'{res['end_time']}'"  if res.get('end_time', '')   not in [None, 'NULL', ''] else "CURRENT_DATETIME()"
            start_time = f"'{res['start_time']}'" if res.get('start_time', '') not in [None, 'NULL', ''] else "CURRENT_DATETIME()"

            values_list.append(f"""(
                '{res['bq_table_name']}', '{res['td_table_name']}', '{res['status']}',
                {bq_cnt}, {td_cnt}, CURRENT_TIMESTAMP(), {end_time}, {start_time},
                'manual_triggered', '{res['writeback_dag_name']}', '{res['bq_table_logical_size_gb']}',
                '{res['load_type']}', '{res['load_functionality']}', '{res['storage_last_modified_time']}'
            )""")

        if values_list:
            bulk_sql = f"""
                INSERT INTO `{BQ_AUDIT_TABLE}`
                (bq_table_name, td_table_name, writeback_status, bq_count, td_count,
                 audit_ts, end_time, start_time, trigger_type, writeback_dag_name,
                 data_size_gb, load_type, load_functionality, table_last_modified_when_wb)
                VALUES {", ".join(values_list)}
            """
            BigQueryHook(gcp_conn_id=BQ_CONN_ID).run_query(sql=bulk_sql, use_legacy_sql=False)
            logger.info(f"Bulk insert complete: {len(values_list)} records processed.")

    mapped_results = bulk_audit_finalize()

    start >> metadata_records >> init_audit_data >> trigger_worker >> mapped_results >> end
