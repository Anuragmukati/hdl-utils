# --- Helper Functions: Formats & SQL Generation ---

def translate_td_format_to_bq(td_format):
    if not td_format:
        return None
    fmt = td_format.upper()
    if fmt == 'DD-MMM-YY':
        return '%d-%b-%y'
    fmt = fmt.replace('YYYY', '%Y').replace('YY', '%y').replace('MM', '%m').replace('DD', '%d')
    return None if fmt in ['%Y-%m-%d', '%Y/%m/%d'] else fmt


def get_td_column_formats(database_name, table_name):
    conn = BaseHook.get_connection(TD_CONN_ID)
    col_formats = {}
    query = (
        f"SELECT ColumnName, ColumnFormat FROM DBC.ColumnsV "
        f"WHERE DatabaseName = '{database_name}' AND TableName = '{table_name}' "
        f"AND ColumnFormat IS NOT NULL"
    )
    try:
        with teradatasql.connect(host=conn.host, user=conn.login, password=conn.password) as con:
            with con.cursor() as cur:
                cur.execute(query)
                for row in cur.fetchall():
                    bq_fmt = translate_td_format_to_bq(row[1].strip())
                    if bq_fmt:
                        col_formats[row[0].strip().upper()] = bq_fmt
    except Exception as e:
        print(f"Warning: Could not fetch formats: {e}")
        return {}
    return col_formats


def build_select_query_and_columns(client, dataset_id, table_id, td_format_map, td_format_map_sel, project_id):
    table = client.get_table(f"{dataset_id}.{table_id}")
    select_exprs, column_names, custom_format_sel_agg = [], [], []
    for field in table.schema:
        col, dtype = field.name, field.field_type.upper()
        column_names.append(col)
        custom_format = td_format_map.get(col.upper())
        custom_format_sel_agg.append(f"{col} AT 'GMT' AS {col}" if dtype in ['DATETIME', 'TIMESTAMP'] else col)

        if dtype == "STRING":
            expr = f"`{project_id}`.edw_data_migration_opsstg_tbls.uni_to_lat(COLLATE(`{col}`, '')) AS `{col}`"
        elif dtype == "DATE":
            expr = (
                f"FORMAT_DATE('{custom_format}', `{col}`) AS `{col}`" if custom_format
                else f"PARSE_DATE('%Y-%m-%d', CAST(`{col}` AS STRING)) AS `{col}`"
            )
        elif dtype in ["DATETIME", "TIMESTAMP"]:
            expr = f"FORMAT_DATETIME('%Y-%m-%d %H:%M:%E*S', `{col}`) AS `{col}`"
        else:
            expr = f"SAFE_CAST(`{col}` AS STRING) AS `{col}`"
        select_exprs.append(expr)

    query = f"SELECT {', '.join(select_exprs)} FROM `{dataset_id}.{table_id}`"
    comma_separated_cols = ", ".join(f'"{c}"' for c in column_names)
    custom_format_sel_agg_final = ", ".join(custom_format_sel_agg)
    return query, comma_separated_cols, custom_format_sel_agg_final


def build_select_query_and_columns_nos(client, td_database, td_table, bq_project, bq_dataset, bq_table):
    table_ref = f"{bq_project}.{bq_dataset}.{bq_table}"
    bq_meta = {field.name.upper(): field.field_type.upper() for field in client.get_table(table_ref).schema}
    conn = BaseHook.get_connection(TD_CONN_ID)
    with teradatasql.connect(host=conn.host, user=conn.login, password=conn.password) as connect:
        with connect.cursor() as cursor:
            cursor.execute(
                f"SELECT ColumnName, ColumnType, DecimalFractionalDigits FROM DBC.ColumnsV "
                f"WHERE DatabaseName = '{td_database.upper()}' AND TableName = '{td_table.upper()}' "
                f"ORDER BY ColumnId;"
            )
            bq_select_exprs, td_select_exprs, col_names = [], [], []
            for row in cursor.fetchall():
                col_name  = row[0].strip()
                col_type  = row[1].strip()
                precision = row[2] if row[2] is not None else 6
                bq_type   = bq_meta.get(col_name.upper(), "")
                col_names.append(col_name)

                if col_type in ['TS', 'SZ', 'TZ'] and precision > 0 and bq_type.upper() in ('TIMESTAMP', 'DATETIME'):
                    td_expr = f"CASE WHEN \"{col_name}\" IS NULL THEN NULL ELSE CAST(\"{col_name}\" AS TIMESTAMP({precision})) END"
                    bq_expr = f"CAST(FORMAT_DATETIME('%E4Y-%m-%d %H:%M:%E{precision}S', `{col_name}`) AS STRING) AS `{col_name}`"
                elif col_type in ['TS', 'SZ', 'TZ'] and precision == 0 and bq_type.upper() in ('TIMESTAMP', 'DATETIME'):
                    td_expr = f"CASE WHEN \"{col_name}\" IS NULL THEN NULL ELSE CAST(\"{col_name}\" AS TIMESTAMP(0)) END"
                    bq_expr = f"CAST(FORMAT_DATETIME('%E4Y-%m-%d %H:%M:%S', `{col_name}`) AS STRING) AS `{col_name}`"
                elif col_type == 'AT' and precision > 0 and bq_type.upper() in ['TIME']:
                    td_expr = f"CASE WHEN \"{col_name}\" IS NULL THEN NULL ELSE CAST(\"{col_name}\" AS TIME({precision})) END"
                    bq_expr = f"CAST(FORMAT_TIME('%H:%M:%E{precision}S', `{col_name}`) AS STRING) AS `{col_name}`"
                elif col_type == 'AT' and precision == 0 and bq_type.upper() in ['TIME']:
                    td_expr = f"CASE WHEN \"{col_name}\" IS NULL THEN NULL ELSE CAST(\"{col_name}\" AS TIME(0)) END"
                    bq_expr = f"CAST(FORMAT_TIME('%H:%M:%S', `{col_name}`) AS STRING) AS `{col_name}`"
                elif col_type == 'F' and bq_type.upper() in ['BIGNUMERIC', 'NUMERIC']:
                    td_expr = f'"{col_name}"'
                    bq_expr = f"CAST(`{col_name}` AS FLOAT64) AS `{col_name}`"
                elif col_type in ['D', 'N'] and bq_type.upper() == 'BIGNUMERIC':
                    td_expr = f'"{col_name}"'
                    bq_expr = f"CAST(`{col_name}` AS NUMERIC) AS `{col_name}`"
                elif col_type in ['D', 'N'] and bq_type.upper() == 'NUMERIC':
                    td_expr = f'"{col_name}"'
                    bq_expr = f"CAST(`{col_name}` AS NUMERIC) AS `{col_name}`"
                elif col_type == 'CO' and bq_type.upper() in ['STRING']:
                    td_expr = f"CAST(\"{col_name}\" AS VARCHAR(30000)) AS \"{col_name}\""
                    bq_expr = f"CAST(`{col_name}` AS STRING) AS `{col_name}`"
                elif col_type in ['I1', 'I2', 'I', 'I4', 'I8'] and bq_type.upper() in ['TIME']:
                    td_expr = f"CAST(\"{col_name}\" AS INTEGER) AS \"{col_name}\""
                    bq_expr = f"CAST(FORMAT_TIME('%H%M%S', `{col_name}`) AS INTEGER) AS `{col_name}`"
                else:
                    td_expr = f'"{col_name}"'
                    bq_expr = f"`{col_name}` AS `{col_name}`"

                td_select_exprs.append(td_expr)
                bq_select_exprs.append(bq_expr)

            return (
                f"SELECT {', '.join(bq_select_exprs)} FROM `{table_ref}`",
                f"SELECT {', '.join(td_select_exprs)} FROM WIRELESS_WRITEBACK.{td_database}_{td_table}_ft",
                ", ".join(col_names)
            )


# --- Incremental Python DAG Deployment ---

def deploy_incremental_dag(full_table_path, schedule, td_tsl_query, gcp_tls_sql_query):
    gcs_hook = GCSHook(gcp_conn_id=BQ_CONN_ID)
    source_blob = "dags/PROJECT_ID_25/Template/writeback_dag_template.txt"
    dataset, table = full_table_path.split('.')
    new_dag_id = f"dg_{project}_bq2td_incremental_load_{dataset}_{table}"
    code = gcs_hook.download(composer_bucket, source_blob).decode('utf-8')
    code = re.sub(r'^project\s*=\s*"[^"]*"', f'project = "{project}"', code, flags=re.MULTILINE)
    code = (code
            .replace('Table_Name = "dataset.table"', f'Table_Name = "{full_table_path}"')
            .replace('dataset="dataset"', f'dataset="{dataset}"')
            .replace('table="table"', f'table="{table}"')
            .replace('td_tsl_query="td_tsl_query"', f'td_tsl_query="{td_tsl_query}"')
            .replace('gcp_tls_sql_query="gcp_tls_sql_query"', f'gcp_tls_sql_query="{gcp_tls_sql_query}"'))
    code = re.sub(r'DAG_ID = f"bq2td_writeback_nos_.*"', f'DAG_ID = "{new_dag_id}"', code)
    if schedule not in [None]:
        code = code.replace('schedule_interval=None', f"schedule_interval={schedule}")
    temp_path = os.path.join(tempfile.gettempdir(), f"{new_dag_id}.py")
    with open(temp_path, "w") as f:
        f.write(code)
    return temp_path
