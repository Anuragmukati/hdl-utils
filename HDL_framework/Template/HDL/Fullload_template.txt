def send_email_report(Status,**kwargs):
    dag_run_id = kwargs['dag'].dag_id
    subject = f"HDL DAG Execution Report: {dag_run_id}"
    recipient_emails = ["user@example.com","user@example.com","user@example.com","user@example.com", "user@example.com"]
    sender_email = "user@example.com"

    # Email content
    html_content = f"""
    <html>
    <body>
        <h1>DAG Execution Completed</h1>
        <p>The DAG <b>{dag_run_id}</b> has {Status}.</p>
    </body>
    </html>
    """

    # Construct the email
    msg = MIMEMultipart("alternative")
    msg["Subject"] = subject
    msg["From"] = sender_email
    msg["To"] = ", ".join(recipient_emails)
    msg.attach(MIMEText(html_content, "html"))

    # Send the email
    smtp_server = "smtp.example.com"
    smtp_port = 25
    with smtplib.SMTP(smtp_server, smtp_port) as server:
        server.ehlo()
        server.starttls()
        server.sendmail(sender_email, recipient_emails, msg.as_string())

    print("Email sent successfully.")

def get_td_tbl_count(td_schema,td_table, cnt_tbl_sql=None):

        #password = access_secret_version("td_pwd")

            host=conn.host,

            password=conn.password
        )

        cursor = connection.cursor()
        if not cnt_tbl_sql:
            cnt_tbl_sql = f"SELECT CAST(COUNT(1) AS BIGINT) FROM {td_schema}.{td_table}"
        print("sql: ",cnt_tbl_sql)

        STOREDAS('parquet')
        MAXOBJECTSIZE('16MB')
        COMPRESSION('SNAPPY')
    ) AS d;
    """

    ti.xcom_push(key='nos_query', value=sql_query)
    ti.xcom_push(key='gcs_path',value=gcs_path)
    ti.xcom_push(key='gcs_bucket',value=gcs_bucket)
    print("Generated Query:", sql_query)

def prepare_gcs_to_tbl_load(**kwargs):
    ti = kwargs['ti']
    gcs_path = ti.xcom_pull(task_ids='generate_nos_query', key='gcs_path')
    gcs_bucket = ti.xcom_pull(task_ids='generate_nos_query', key='gcs_bucket')
    stg_db = ti.xcom_pull(task_ids='fetch_metadata', key='stg_db')
    stg_tbl = ti.xcom_pull(task_ids='fetch_metadata', key='stg_tbl')
    tgt_db = ti.xcom_pull(task_ids='fetch_metadata', key='tgt_db')
    tgt_tbl = ti.xcom_pull(task_ids='fetch_metadata', key='tgt_tbl')

    hook = BigQueryHook(gcp_conn_id=GCP_CONN_DO_ID, use_legacy_sql=False)
    gcs_hook = BigQueryHook(gcp_conn_id=GCS_CONN_DO_ID, use_legacy_sql=False)
    gcs_client = gcs_hook.get_client()

    # Check if files exist in the GCS path - use credentials from the BigQuery client
    storage_client = storage.Client(
        project=bq_prod_project,
        credentials=gcs_client._credentials
    )
    bucket = storage_client.bucket(gcs_bucket)
    blobs = list(bucket.list_blobs(prefix=gcs_path))

    if not blobs:
        print(f"No files found in GCS path: gs://{gcs_bucket}/{gcs_path}. Skipping further steps.")
        truncate_sql = f"TRUNCATE TABLE {bq_prod_project}.{tgt_db}.{tgt_tbl};"
        conn = hook.get_conn()
        cursor = conn.cursor()
        try:
            cursor.execute(truncate_sql)
            print(f"Truncated table {tgt_db}.{tgt_tbl} in BigQuery.")
        except Exception as e:
            print(f"Failed to truncate table {tgt_db}.{tgt_tbl}: {e}")
            raise AirflowFailException(f"Failed to truncate table {tgt_db}.{tgt_tbl}: {e}")
        
        updt_sql("SKIPPED", error_message="No files found in GCS path. Source has zero count.", **kwargs)
        raise AirflowSkipException("No files found in GCS path. Skipping the rest of the steps.")
    else:
        print(f"Files found in GCS path: gs://{gcs_bucket}/{gcs_path}. Proceeding with the next steps.")

def clean_error_msg(error_msg):
    msg=str(error_msg)
    msg=re.sub(r"[\n\r\t]+"," ",msg)
    msg=re.sub(r"[\"']","",msg)
    msg=re.sub(r"\s{2,}"," ",msg)
    return msg.strip()

def fail_task(context):
    task_instance=context.get('task_instance')
    exception=context.get('exception')
    error=clean_error_msg(str(task_instance)+" "+str(exception))+"|Second run with parquet"
    updt_sql("Dag_Failed",error_message=error,**context)
    send_email_report("Dag Got Failed",**context)
    print("DAG execution failed!",error)
