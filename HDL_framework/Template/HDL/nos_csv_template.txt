def get_td_tbl_count(td_schema,td_table, cnt_tbl_sql=None):
    conn=BaseHook.get_connection(functional_id_used)
    try:
        print("**** Getting td row count")
        connection = teradatasql.connect(
            host=conn.host,
            user=conn.login,
            password=conn.password
        )

        cursor = connection.cursor()
        if not cnt_tbl_sql:
            cnt_tbl_sql = f"SELECT CAST(COUNT(1) AS BIGINT) FROM {td_schema}.{td_table}"
        print("sql: ",cnt_tbl_sql)
        cursor.execute(cnt_tbl_sql)
        count = cursor.fetchone()[0]
        cursor.close()
        connection.close()
        return count

    except Exception as e:
        print(f"**** Error while getting td row count for table {td_schema}.{td_table}: {str(e)}")
        return -1


def get_bq_tbl_count(db_nm,tbl_nm,conn_id=None,project=None,cnt_tbl_sql=None):

    hook = BigQueryHook(gcp_conn_id=_conn, use_legacy_sql=False)
    conn = hook.get_conn()
    cursor = conn.cursor()
    try:
        cursor.execute(cnt_tbl_sql)
    except Exception as e:
        print(f"Count Query Got Failed: {e}")
        return -1
    rows = cursor.fetchone()
    print(rows)
    if rows:
        return rows[0]
    else:
        print("Table not found")
        return -1

def fetch_metadata(**kwargs):
    #### Sql for fetching details from Queue table if any
    queue_tbl_sql="select table_uid,app_name,ple_load,prod_load,dag_run_id from "+str(HDL_project)+"."+str(Metadata_db)+"."+str(Queue_Table)+" where STATUS!='INPROGRESS' and app_name = '"+str(App_Name)+"' and cast(queue_number as int64)="+str(Sequence_id)+" order by INSERT_DT limit 1"
    hook = BigQueryHook(gcp_conn_id=hdl_source_connection_id, use_legacy_sql=False)
    conn = hook.get_conn()
    cursor = conn.cursor()
    try:
        cursor.execute(queue_tbl_sql)
    except:
        print("Queue Table Query Got Failed")
        raise AirflowFailException("Queue Table Query Got Failed")
    Queue_Tbl_details=cursor.fetchone()
    print(Queue_Tbl_details)
    #### Taking Data from Queue Table
    if Queue_Tbl_details:
        tbl_uid=Queue_Tbl_details[0]
        app_name=Queue_Tbl_details[1]
        ple_load=Queue_Tbl_details[2]
        prod_load=Queue_Tbl_details[3]
        dag_run_id=Queue_Tbl_details[4]

        raise AirflowSkipException("No Table To load in Queue") 
    
    #### Updating the Queue Table 
    updt_queue_tbl(tbl_uid,app_name,dag_run_id)
    
    #### Fetching Metadata Details depending on Queue Table Details.
    metadata_sql="select * from "+str(HDL_project)+"."+str(Metadata_db)+"."+str(Metadata_tbl)+" where  is_table_active is True and lower(Table_uid)=lower('"+str(tbl_uid)+"') and Lower(App_Name) like lower('"+str(app_name)+"')"
    try:
        cursor.execute(metadata_sql)
    except:
        print("Metadata Table Query Got Failed")
        raise AirflowFailException("Metadata Table Query Got Failed")
    Metadata_details=cursor.fetchone()
    print(Metadata_details)
    cur_time=datetime.now(tz=pendulum.timezone("America/New_York"))
    if Metadata_details:
        tbl_uid=Metadata_details[1]
        src_db=Metadata_details[2]
        src_tbl=Metadata_details[3]
        stg_db=Metadata_details[4]
        stg_tbl=Metadata_details[5]
        tgt_db=Metadata_details[6]
        tgt_tbl=Metadata_details[7]
        app_name=Metadata_details[13]
        stg_load_flg=Metadata_details[21]
        pelican_val_flg=Metadata_details[22]
        fetch_query=Metadata_details[23]
    else:
        print("No Details not Found in Metadata table",tbl_uid)
        del_queue_tbl(tbl_uid,app_name,dag_run_id)
        raise AirflowSkipException("No Details not Found in Metadata table for ",tbl_uid) 
    #### Inserting the Audit Entry
    td_count=get_td_tbl_count(src_db,src_tbl)
    ti = kwargs['ti']
    bq_count=get_bq_tbl_count(tgt_db,tgt_tbl,conn_id=active_conn,project=active_project)
    ti.xcom_push(key='metadata', value={
        'tbl_uid': tbl_uid,
        'src_db': src_db,
        'src_tbl': src_tbl,
        'stg_db': stg_db,
        'stg_tbl': stg_tbl,
        'tgt_db': tgt_db,
        'tgt_tbl': tgt_tbl,
        'app_name': app_name,
        'stg_load_flg': stg_load_flg,
        'pelican_val_flg': pelican_val_flg,
        'ple_load': ple_load,
        'prod_load': prod_load,
        'active_project': active_project,
        'fetch_query': fetch_query,
        'functional_id_used': functional_id_used,
        'start_time': str(cur_time),
        'dag_run_id': dag_run_id,
        'td_count': td_count,
        'bq_count': bq_count,
    })
    updt_sql("STARTED",**kwargs)
    if td_count==0:
        updt_sql("SKIPPED",error_message='Source have Zero Count',**kwargs)
        print("Table skipped due to zero count")
        del_queue_tbl(tbl_uid,app_name,dag_run_id)
        raise AirflowSkipException("Table skipped due to zero count")
    elif td_count==bq_count:
        updt_sql("SKIPPED",error_message='Source and Target(ple) count Are Matching.',**kwargs)
        print("Table skipped due to count matching")
        del_queue_tbl(tbl_uid,app_name,dag_run_id)
        raise AirflowSkipException("TD and BQ Table count is same, data refreshed to ple")

def generate_nos_query(**kwargs):
    #### Hitting Teradata and extracting data into GCS files
    ti = kwargs['ti']
    meta = ti.xcom_pull(task_ids='fetch_metadata', key='metadata') or {}
    db_nm = meta.get('tgt_db')
    tbl_nm = meta.get('tgt_tbl')
    fetch_query = meta.get('fetch_query')
    dt=kwargs['execution_date'].strftime('%Y%m%dT%H%M%S')
    gcs_path = "nos/input/"+str(db_nm)+'_'+str(tbl_nm)+'/'+str(dt)
    #### Creating NOS SQL Query for Reading Data.
    sql_query = f"""
    SELECT * FROM WRITE_NOS (
        ON ( {fetch_query} )
        USING
        LOCATION('/gs/storage-ENDPOINT.p.googleapis.com/{gcs_bucket}/{gcs_path}/')
        AUTHORIZATION({TD_Auth_used})
        STOREDAS('CSV')
        ROWFORMAT('{{"field_delimiter":"|","record_delimiter":"\\n"}}')
        MAXOBJECTSIZE('16MB')
        COMPRESSION('GZIP')
    ) AS d;
    """

    ti.xcom_push(key='nos_info', value={
        'nos_query': sql_query,
        'gcs_path': gcs_path,
        'gcs_bucket': gcs_bucket,
    })
    print("Generated Query:", sql_query)

def clean_error_msg(error_msg):
    msg=str(error_msg)
    msg=re.sub(r"[\n\r\t]+"," ",msg)
    msg=re.sub(r"[\"']","",msg)
    msg=re.sub(r"\s{2,}"," ",msg)
    return msg.strip()

def fail_task(context):
    ti=context['ti']
    meta = ti.xcom_pull(task_ids='fetch_metadata', key='metadata') or {}
    db_nm = meta.get('tgt_db')
    tbl_nm = meta.get('tgt_tbl')

    prod_load = meta.get('prod_load')
    app_nm = meta.get('app_name')
    dag_run_id = meta.get('dag_run_id')
    tbl_uid = meta.get('tbl_uid')
    task_instance=context.get('task_instance')
    exception=context.get('exception')
    error=clean_error_msg(str(task_instance)+" "+str(exception))
    task=str(((str(task_instance).split('>')[0]).split('.')[1]).split(' ')[0])
    ##### 2nd try with parquet dag ##############
    if task.lower() in ('nos_to_gcs_load',f'gcs_to_bq_{env.lower()}_load'):
        Queue_Table_sql=f"insert into {HDL_project}.{Metadata_db}.{Queue_Table}_p (table_uid,table_dataset,table_name,app_name,ple_load,prod_load,dag_run_id,queue_number,second_run) values('{db_nm}__{tbl_nm}','{db_nm}','{tbl_nm}','{app_nm}',{ple_load},{prod_load},'{dag_run_id}', '{Sequence_id}',True)"
        hook = BigQueryHook(gcp_conn_id=hdl_source_connection_id, use_legacy_sql=False)
        conn = hook.get_conn()
        cursor = conn.cursor()
        try:
            cursor.execute(Queue_Table_sql)
            del_queue_tbl(tbl_uid,app_nm,dag_run_id)
        except:
            print("2nd Attempt Also Failed")
            updt_sql("Dag_Failed",error_message=str(error)+"|2nd Attempt also Failed",**context)
            raise AirflowFailException("2nd Attempt also Failed")
            
    else:
        updt_sql("Dag_Failed",error_message=error,**context)
        print("DAG execution failed!",error)
